const getWeekCtx = dv => {
  const cur = dv?.current?.() || {};
  const af = app.workspace.getActiveFile();
  const curPath = cur.file?.path || af?.path || "";
  const curName = cur.file?.name || af?.basename || "";
  const curTFile = curPath ? app.vault.getAbstractFileByPath(curPath) : af;

  const probe = `${curName}${curPath}`;
  const yHit = probe.match(/Diary\/(\d{4})\//) || probe.match(/(20\d{2})/);
  let Y = yHit ? +yHit[1] : isoWeekInfo(new Date()).year;
  let WK = 0;

  const mW = probe.match(/(?:^|[^A-Za-z])[Ww](\d{1,2})(?!\d)/);
  const m8 = probe.match(/(20\d{2})\D?(\d{2})\D?(\d{2})(?!\d)/);
  if (mW) {
    WK = +mW[1];
  } else if (m8) {
    const info = isoWeekInfo(new Date(+m8[1], +m8[2] - 1, +m8[3]));
    Y = info.year;
    WK = info.week;
  }
  if (!(WK >= 1 && WK <= 53)) {
    const nowInfo = isoWeekInfo(new Date());
    Y = nowInfo.year;
    WK = nowInfo.week;
  }

  const jan4 = new Date(Y, 0, 4);
  const MON = new Date(jan4.getTime());
  MON.setDate(jan4.getDate() - ((jan4.getDay() || 7) - 1) + (WK - 1) * 7);
  MON.setHours(0, 0, 0, 0);

  const dayOf = i => {

  };

};

const clean = s => String(s || "")
  .replace(/%%[\s\S]*?%%/g, "")
  .replace(/<!--[\s\S]*?-->/g, "")
  .replace(/^[ \t]*[-*_]{3,}[ \t]*$/gm, "")
  .replace(/!\[\[[^\]]*\]\]/g, "")
  .replace(/\[\[([^\]|]*)\|([^\]]*)\]\]/g, "$2")
  .replace(/\[\[([^\]]*)\]\]/g, "$1")
  .replace(/^[ \t]*#{1,6}[ \t]*/gm, "")
  .replace(/\*\*/g, "")
  .split("|").join("／")
  .replace(/\s+/g, " ")
  .trim();

const cut = (s, n) => {
  const t = clean(s);
  return t.length > n ? t.slice(0, n) + "…" : t;
};

const secOf = (block, kw) => {
  const re = new RegExp(`^###[ \\t]*[^\\n\\w\\u4e00-\\u9fa5]*[ \\t]*${kw}[^\\n]*$`, "m");
  const m = re.exec(block);
  if (!m) return "";
  const rest = block.slice(m.index + m[0].length);
  const next = rest.search(/^#{2,3}[ \t]/m);
  return next < 0 ? rest : rest.slice(0, next);
};

const parseDiary = raw => {
  const stripped = String(raw || "")
    .replace(/\r\n?/g, "\n")
    .replace(/%%[\s\S]*?%%/g, "")
    .replace(/<!--[\s\S]*?-->/g, "");

  return stripped
    .split(/^##[ \t]*📌[ \t]*/m)
    .slice(1)
    .map(b => {
      const head = (b.split("\n")[0] || "").trim();
      const score = (b.match(/^[ \t]*[-*]?[ \t]*(?:\*\*)?重要度(?:\*\*)?[ \t]*::[ \t]*(\d+(?:\.\d+)?)/m) || [])[1] || "";

      const analysisSec = secOf(b, "分析");
      const mind = (analysisSec.match(/^[ \t]*[-*]?[ \t]*(?:\*\*)?洞察(?:\*\*)?[ \t]*::[ \t]*(.*)$/m) || [])[1] || "";

      const planSec = secOf(b, "计划");
      const hasWW = /^[ \t]*[-*]?[ \t]*(?:\*\*)?(?:What|When)(?:\*\*)?[ \t]*::/im.test(planSec);
      let act = "";
      if (hasWW) {
        const w1 = (planSec.match(/^[ \t]*[-*]?[ \t]*(?:\*\*)?What(?:\*\*)?[ \t]*::[ \t]*(.*)$/im) || [])[1] || "";
        const w2 = (planSec.match(/^[ \t]*[-*]?[ \t]*(?:\*\*)?When(?:\*\*)?[ \t]*::[ \t]*(.*)$/im) || [])[1] || "";
        act = [clean(w1), clean(w2)].filter(Boolean).join(" ｜ ");
        if (/^原文未写，待补 ｜ 原文未写，待补$/.test(act)) act = "原文未写，待补";
      } else {
        act = clean(planSec.replace(/^[ \t]*[-*]?[ \t]*[^:\n]{1,12}::[ \t]*$/gm, ""));
      }

      const res = clean(secOf(b, "简述"))
        .replace(/^第[-一二三四五六七八九十百\d]+件事[：:]\s*/, "")
        .trim();

      return { head, score, mind, act, res };
    });
};

const findDiaryFile = dateObj => {
  const d8 = fmtYMD(dateObj);
  const prefix = `Diary/${dateObj.getFullYear()}/`; const files = app.vault.getMarkdownFiles(); return ( files.find(f => f.path.startsWith(prefix) && f.basename === d8) \vert{}\vert{} files.find(f => f.path.startsWith(prefix) && f.basename.startsWith(d8)) \vert{}\vert{} null ); };  const loadDay = async (ctx, dayIdx) => { const dateObj = ctx.dayOf(dayIdx); const d8 = fmtYMD(dateObj); const file = findDiaryFile(dateObj); if (!file) return { dateObj, d8, file: null, evs: [] };  const key = `${file.path}:${file.stat?.mtime \vert{}\vert{} 0}`; if (!FILE_CACHE.has(key)) { if (FILE_CACHE.size > 120) FILE_CACHE.clear(); FILE_CACHE.set( key, app.vault.cachedRead(file).then(parseDiary).catch(() => []) ); } const evs = await FILE_CACHE.get(key); return { dateObj, d8, file, evs }; };  const makeLink = (target, label, sourcePath) => { const a = document.createElement("a"); a.className = "internal-link"; a.textContent = label; a.setAttribute("data-href", target); a.setAttribute("href", target); a.addEventListener("click", evt => { evt.preventDefault(); app.workspace.openLinkText(target, sourcePath, evt.ctrlKey \vert{}\vert{} evt.metaKey); }); return a; };  const calcMerges = counts => { const merges = []; let rowNo = 1; for (let i = 0; i < 7; i++) { const n = counts[i]; const start = rowNo + 1; rowNo += n; if (n > 1) merges.push(`A${start}:A${rowNo}`);

const fillCell = async (el, dv, kind, dayIdx, evIdx) => {
  try {
    const ctx = getWeekCtx(dv);

    if (kind === "cover") {
      const days = await Promise.all([0, 1, 2, 3, 4, 5, 6].map(i => loadDay(ctx, i)));
      const nDay = days.filter(d => d.file).length;
      const nEv = days.reduce((s, d) => s + d.evs.length, 0);
      el.replaceChildren(document.createTextNode(`${nDay} / 7 天有日记｜事件 ${nEv} 条`));

      if (ctx.curTFile) {
        const curText = await app.vault.cachedRead(ctx.curTFile);
        const curCounts = [0, 0, 0, 0, 0, 0, 0];
        for (const m of curText.matchAll(/WR\(dv,\s*['"]ev['"],\s*(\d+),\s*(\d+)\)/g)) {
          const di = +m[1];
          if (di >= 0 && di < 7) curCounts[di]++;
        }
        const expCounts = days.map(d => Math.max(1, d.evs.length));
        const expMerges = JSON.stringify(calcMerges(expCounts));
        const curMergeHit = curText.match(/<!--\s*gogoga-table:\s*(\{[^\n]*\})\s*-->/);
        let mergeMismatch = false;
        if (curMergeHit) {
          try {
            mergeMismatch = JSON.stringify(JSON.parse(curMergeHit[1]).merge) !== expMerges;
          } catch (_) {
            mergeMismatch = true;
          }
        }
        if (expCounts.some((n, i) => curCounts[i] !== n) || mergeMismatch) {
          const btn = document.createElement("button");
          btn.textContent = "🔄 同步表格行数";
          btn.style.cssText =
            "margin-left:10px;padding:1px 8px;font-size:0.85em;cursor:pointer;border-radius:4px;";
          btn.addEventListener("click", () => syncTable(ctx.curTFile, days));
          el.appendChild(btn);
        }
      }
      return;
    }

    const day = await loadDay(ctx, dayIdx);
    const ev = day.evs[evIdx];

    if (kind === "ev") {
      if (!day.file) {
        el.textContent = evIdx === 0 ? "暂无日记记录" : "—";
      } else if (!day.evs.length) {
        el.textContent = evIdx === 0 ? "有日记，无 📌 事件" : "—";
      } else if (!ev) {
        el.textContent = "—";
      } else {
        const base = day.file.path.replace(/\.md$/, "");
        el.replaceChildren(makeLink(`${base}#📌 ${ev.head}`, cut(ev.head, 40), ctx.curPath));
      }
      return;
    }

    if (kind === "mind" || kind === "act" || kind === "res") {
      el.textContent = ev ? (cut(ev[kind], 150) || "—") : "—";
      return;
    }

    if (kind === "notes") {
      if (!day.file) {
        el.textContent = "—";
      } else if (!day.evs.length) {
        if (evIdx === 0) {
          const base = day.file.path.replace(/\.md$/, "");
          el.replaceChildren(
            makeLink(base, day.d8, ctx.curPath),
            document.createTextNode(" · 📌×0")
          );
        } else {
          el.textContent = "—";
        }
      } else if (!ev) {
        el.textContent = "—";
      } else {
        const base = day.file.path.replace(/\.md$/, "");
        el.replaceChildren(
          makeLink(base, day.d8, ctx.curPath),
          document.createTextNode(` · ★${ev.score \vert{}\vert{} "?"} · 📌×${day.evs.length}`)
        );
      }
    }
  } catch (err) {
    el.textContent = "⚠" + (err?.message || "err");
  }
};

window.WR = (dv, kind, dayIdx = 0, evIdx = 0) => {
  const el = document.createElement("span");
  el.textContent = "…";
  fillCell(el, dv, kind, dayIdx, evIdx);
  return el;
};
