
```dataviewjs
// ══════════════════════════════════════════════════════════════════════
// 周报渲染器：读取本周 7 天日记，按事件生成 HTML 表格
// 日期列用 rowspan 合并；其余列逐事件独立显示。
// 备注：修改 NOTE；总结：修改 SUMMARY。
// ══════════════════════════════════════════════════════════════════════

// 备注键格式："星期索引:事件索引"，索引从 0 开始。
// 状态词建议只使用：新起 / 延续 / 调整 / 转折 / 收束 / 单独。
const NOTE = {
  // "0:0": "新起",
  // "0:1": "单独",
  // "1:0": "延续",
};

// 每周手动填写；表格底部会显示这段文字。
const SUMMARY = "主线：；最大变化：；可核对的结果：；没解决的事：；下周只做一件事：";

const p2 = n => String(n).padStart(2, "0");
const DAY_NAMES = ["周一", "周二", "周三", "周四", "周五", "周六", "周日"];
const MS_PER_DAY = 86400000;

// 返回某个本地日期对应的 ISO 周年、周号。
const isoInfo = date => {
  const u = new Date(Date.UTC(date.getFullYear(), date.getMonth(), date.getDate()));
  u.setUTCDate(u.getUTCDate() + 4 - (u.getUTCDay() || 7));
  const isoYear = u.getUTCFullYear();
  const jan1 = new Date(Date.UTC(isoYear, 0, 1));
  const isoWeek = Math.ceil(((u - jan1) / MS_PER_DAY + 1) / 7);
  return { year: isoYear, week: isoWeek };
};

// ── 识别周报对应的年份和周次 ──────────────────────────────────────────
const current = dv.current() || {};
const activeFile = app.workspace && app.workspace.getActiveFile
  ? app.workspace.getActiveFile()
  : null;
const identity = [
  current.file && current.file.name,
  current.file && current.file.path,
  activeFile && activeFile.basename,
  activeFile && activeFile.path,
].filter(Boolean).join(" ");

const diaryYearMatch = identity.match(/(?:^|[/\\])Diary[/\\](20\d{2})(?:[/\\]|$)/i);
const anyYearMatch = identity.match(/(20\d{2})/);
let YEAR = diaryYearMatch ? Number(diaryYearMatch[1])
  : anyYearMatch ? Number(anyYearMatch[1])
  : isoInfo(new Date()).year;

let WEEK = 0;
const dateMatch = identity.match(/(20\d{2})\D?(\d{2})\D?(\d{2})(?!\d)/);
const weekMatch = identity.match(/[Ww](\d{1,2})(?!\d)/);

if (dateMatch) {
  const dateFromName = new Date(
    Number(dateMatch[1]), Number(dateMatch[2]) - 1, Number(dateMatch[3])
  );
  const info = isoInfo(dateFromName);
  YEAR = info.year;
  WEEK = info.week;
} else if (weekMatch) {
  WEEK = Number(weekMatch[1]);
  const explicitWeekYear = identity.match(/(20\d{2})\D*[Ww]\d{1,2}/);
  if (explicitWeekYear) YEAR = Number(explicitWeekYear[1]);
} else {
  const info = isoInfo(new Date());
  YEAR = info.year;
  WEEK = info.week;
}

if (!(WEEK >= 1 && WEEK <= 53)) {
  const info = isoInfo(new Date());
  YEAR = info.year;
  WEEK = info.week;
}

// ISO 周一：以该 ISO 年 1 月 4 日所在周的周一为基准。
const jan4 = new Date(YEAR, 0, 4);
const monday = new Date(jan4);
monday.setDate(jan4.getDate() - ((jan4.getDay() || 7) - 1) + (WEEK - 1) * 7);
monday.setHours(0, 0, 0, 0);

const dayDate = index => {
  const d = new Date(monday);
  d.setDate(d.getDate() + index);
  return d;
};
const dateKey = date => `${date.getFullYear()}${p2(date.getMonth() + 1)}${p2(date.getDate())}`;
const monthDay = date => `${p2(date.getMonth() + 1)}/${p2(date.getDate())}`;
const fullDate = date => `${date.getFullYear()}-${p2(date.getMonth() + 1)}-${p2(date.getDate())}`;

// ── 查找与解析日记 ────────────────────────────────────────────────────
const markdownFiles = app.vault.getMarkdownFiles();
const fileOf = key => markdownFiles.find(file =>
  file.path.startsWith(`Diary/${YEAR}/`) && file.basename.startsWith(key)
);

// 删除 Obsidian 注释、HTML 注释、图片嵌入，并把常见 Markdown 写法转成普通文本。
const clean = value => String(value ?? "")
  .replace(/%%[\s\S]*?%%/g, " ")
  .replace(/<!--[\s\S]*?-->/g, " ")
  .replace(/!\[\[[^\]]*\]\]/g, " ")
  .replace(/\[\[([^\]|]*)\|([^\]]*)\]\]/g, "$2")
  .replace(/\[\[([^\]]*)\]\]/g, "$1")
  .replace(/^[ \t]*#{1,6}[ \t]*/gm, "")
  .replace(/\*\*/g, "")
  .split("|").join("／")
  .replace(/\s+/g, " ")
  .trim();

const cut = (value, limit = 150) => {
  const text = clean(value);
  return text.length > limit ? text.slice(0, limit) + "…" : text;
};

// 取指定三级标题的正文；只由下一个二级/三级标题结束，#### 不会截断。
const sectionOf = (eventBlock, headingText) => {
  const marker = `### ${headingText}`;
  const start = eventBlock.indexOf(marker);
  if (start < 0) return "";
  const rest = eventBlock.slice(start + marker.length);
  const nextHeading = rest.search(/^#{2,3}[ \t]+/m);
  return nextHeading < 0 ? rest : rest.slice(0, nextHeading);
};

const parseEvents = rawText => String(rawText ?? "")
  .replace(/%%[\s\S]*?%%/g, "")
  .split(/^##[ \t]*📌[ \t]*/m)
  .slice(1)
  .map(block => {
    const head = (block.split("\n")[0] || "").trim();

    const score = (block.match(/^[ \t]*[-*]?[ \t]*重要度::[ \t]*(\d+)/m) || [])[1] || "";
    const analysis = sectionOf(block, "🔍 分析");
    const mind = (analysis.match(/^[ \t]*[-*]?[ \t]*洞察::[ \t]*(.*)$/m) || [])[1] || "";

    const plan = sectionOf(block, "🎯 计划");
    const what = (plan.match(/^[ \t]*[-*]?[ \t]*What::[ \t]*(.*)$/mi) || [])[1] || "";
    const when = (plan.match(/^[ \t]*[-*]?[ \t]*When::[ \t]*(.*)$/mi) || [])[1] || "";
    const actionParts = [clean(what), clean(when)].filter(Boolean);
    let action = actionParts.join(" ｜ ");
    if (action === "原文未写，待补 ｜ 原文未写，待补") action = "原文未写，待补";

    const brief = sectionOf(block, "📝 简述");
    const result = clean(brief).replace(/^第[-一二三四五六七八九十\d]+件事[：:]\s*/, "");

    return { head, score, mind: clean(mind), action, result };
  });

// ── 读取本周七天 ──────────────────────────────────────────────────────
const days = [];
for (let i = 0; i < 7; i++) {
  const key = dateKey(dayDate(i));
  const file = fileOf(key);
  let events = [];
  let readError = false;

  if (file) {
    try {
      events = parseEvents(await dv.io.load(file.path));
    } catch (error) {
      readError = true;
      console.error(`周报读取日记失败：${file.path}`, error);
    }
  }

  days.push({ index: i, key, file, hasDiary: Boolean(file), events, readError });
}

const diaryDays = days.filter(day => day.hasDiary).length;
const eventCount = days.reduce((sum, day) => sum + day.events.length, 0);

// ── HTML 安全转义与 Obsidian 内部链接 ──────────────────────────────────
const escapeHtml = value => String(value ?? "")
  .replace(/&/g, "&amp;")
  .replace(/</g, "&lt;")
  .replace(/>/g, "&gt;")
  .replace(/"/g, "&quot;")
  .replace(/'/g, "&#39;");

const internalLink = (target, label) => {
  const safeTarget = escapeHtml(target);
  return `<a class="internal-link" data-href="${safeTarget}" href="${safeTarget}">${escapeHtml(label)}</a>`;
};

// ── 绘制表格 ──────────────────────────────────────────────────────────
const cellStyle = "border:1px solid var(--background-modifier-border);padding:6px 8px;vertical-align:top;white-space:normal;";
const headerStyle = cellStyle + "background:var(--background-secondary);font-weight:600;text-align:left;";
const headers = ["日期", "事件（一句话，链到日记那一条）", "内心", "行动", "结果", "备注", "笔记"];
const minWidths = ["88px", "210px", "180px", "180px", "220px", "90px", "170px"];

let html = '<table style="border-collapse:collapse;width:100%;min-width:1100px;font-size:0.92em;line-height:1.55;"><thead><tr>';
for (let i = 0; i < headers.length; i++) {
  html += `<th style="${headerStyle}min-width:${minWidths[i]};">${escapeHtml(headers[i])}</th>`;
}
html += "</tr></thead><tbody>";

for (const day of days) {
  const events = day.events;
  const dateLabel = `${DAY_NAMES[day.index]} ${monthDay(dayDate(day.index))}`;

  if (events.length === 0) {
    const eventMessage = day.readError
      ? "日记读取失败，请检查 DataviewJS 控制台"
      : day.hasDiary
        ? "有日记，但没有 📌 事件"
        : "暂无日记记录";
    html += "<tr>";
    html += `<td style="${cellStyle}">${escapeHtml(dateLabel)}</td>`;
    html += `<td style="${cellStyle}">${escapeHtml(eventMessage)}</td>`;
    html += `<td style="${cellStyle}">—</td><td style="${cellStyle}">—</td><td style="${cellStyle}">—</td>`;
    html += `<td style="${cellStyle}">—</td><td style="${cellStyle}">—</td>`;
    html += "</tr>";
    continue;
  }

  events.forEach((event, eventIndex) => {
    html += "<tr>";

    // 只有日期列合并；一个事件时 rowspan=1，多个事件时按事件数合并。
    if (eventIndex === 0) {
      html += `<td rowspan="${events.length}" style="${cellStyle}">${escapeHtml(dateLabel)}</td>`;
    }

    const title = event.head || "未命名事件";
    const eventTarget = `${day.key}#📌 ${title}`;
    html += `<td style="${cellStyle}">${internalLink(eventTarget, cut(title, 40))}</td>`;
    html += `<td style="${cellStyle}">${escapeHtml(cut(event.mind, 150) || "—")}</td>`;
    html += `<td style="${cellStyle}">${escapeHtml(cut(event.action, 150) || "—")}</td>`;
    html += `<td style="${cellStyle}">${escapeHtml(cut(event.result, 150) || "—")}</td>`;
    html += `<td style="${cellStyle}">${escapeHtml(NOTE[`${day.index}:${eventIndex}`] || "—")}</td>`;

    const dayLink = internalLink(day.key, day.key);
    html += `<td style="${cellStyle}">${dayLink} · ★${escapeHtml(event.score || "?")} · 📌×${events.length}</td>`;
    html += "</tr>";
  });
}

// 总结行横向合并七列。修改上方 SUMMARY 常量即可填写本周总结。
html += `<tr><td colspan="7" style="${cellStyle}background:var(--background-secondary);"><strong>总结：</strong>${escapeHtml(SUMMARY)}</td></tr>`;
html += "</tbody></table>";

// ── 输出标题、周期、覆盖统计和表格 ────────────────────────────────────
const weekStart = dayDate(0);
const weekEnd = dayDate(6);
dv.header(1, `W${WEEK} · ${fullDate(weekStart)} ~ ${fullDate(weekEnd)}`);
dv.paragraph(
  `周期：**${fullDate(weekStart)}（周一）～ ${fullDate(weekEnd)}（周日）**` +
  ` ｜ ISO 第 **${WEEK}** 周 ｜ 覆盖：**${diaryDays} / 7 天有日记**` +
  ` ｜ 事件 **${eventCount} 条**`
);

const tableWrapper = dv.container.createEl("div");
tableWrapper.style.overflowX = "auto";
tableWrapper.style.width = "100%";
tableWrapper.innerHTML = html;

dv.paragraph("备注列通过本代码块顶部的 NOTE 手写；事件、内心、行动、结果、笔记及日期合并均自动生成。日记更新后重新渲染本页面即可同步。 ");
```
