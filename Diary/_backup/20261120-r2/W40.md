```dataviewjs
// ═══════════════ ⚙️ 配置区（每周只改这一段） ═══════════════
const WEEK_NO = "W40";
const START   = "20260928";   // 周一
const END     = "20261004";   // 周日
const FOLDER  = "Diary/2026";
// 事件 ID 映射：日期 → [[事件ID, 一句话]]（唯一的人工输入，改动只需动这里）
const EVT = {
  "20260929": [["event-2026-004", "下载「青藤之恋」→ 匹配并加微信 → 决定每天最多半小时"], ["event-2026-003", "精读重构：中文反译回德语 + 复听，自评效果大幅提升"]],
  "20261001": [["event-2026-004", "加微信后的聊天复盘：把失败改判为交互结构问题（属 10 月，计入 M10）"]]
};

// ═══════════════ 🔧 工具 ═══════════════
const WD = ["周一","周二","周三","周四","周五","周六","周日"];
const CIRC = ["①","②","③","④","⑤","⑥","⑦","⑧"];
const RE_MAIN = /(\d+(?:\.\d+)?\s*个?\s*(?:小时|分钟|元|块|次|步|篇)|\d+\s*\/\s*100)/g;
const RE_ALT  = /(\d+\s*Uhr|\b(?:ein|zwei|drei|vier|fünf|sechs|sieben)\s+Stunden?\b|über eine halbe Stunde)/g;
const MUTE = '<span style="opacity:.55">';

const stripC = raw => String(raw ?? "").replace(/%%[\s\S]*?%%/g, "").replace(/<!--[\s\S]*?-->/g, "");

// 只读解析：事件分隔符 = "## 📌"，字段 = "key:: value"，章节 = "### xxx"，%% 提示语全部剔除
function parseDiary(raw){
  const out = [];
  for (const b of stripC(raw).split(/^##\s*📌\s*/m).slice(1)){
    const lines = b.split("\n");
    const ev = { title:(lines[0]||"").trim(), rating:0, themes:[], persons:[], fields:{}, plan:{}, free:{} };
    let sec = "";
    for (const line of lines.slice(1)){
      const h = line.match(/^###\s+([^#\s][^\n]*)$/);
      if (h){ sec = h[1].replace(/[^\u4e00-\u9fa5A-Za-z]/g, ""); continue; }
      if (/^#{1,2}\s/.test(line)){ sec = ""; continue; }
      const f = line.match(/^\s*-?\s*([\w\u4e00-\u9fa5]+)::\s*(.*)$/);
      if (f){
        const k = f[1], v = (f[2]||"").trim();
        if (k === "重要度"){ ev.rating = Math.min(5, parseInt(v)||0); continue; }
        if (k === "主题"){ ev.themes = v.split(/[\s,，]+/).filter(t => t.startsWith("#")); continue; }
        if (k === "人物"){ if (v) ev.persons.push(v); continue; }
        if (!v) continue;
        if (["What","Who","When","Where","How","预算","预期"].includes(k)) ev.plan[k] = v;
        else ev.fields[k] = ev.fields[k] ? ev.fields[k] + "；" + v : v;
        continue;
      }
      const t = line.trim();
      if (sec && t && !/^[-*|`>]/.test(t)) ev.free[sec] = (ev.free[sec]||"") + t + " ";
    }
    out.push(ev);
  }
  return out;
}

// 结果列：优先取可验证的中文量词（小时/元/分/分数），没有再退到德语时长与钟点
function pickNumbers(evs){
  const grab = re => {
    const out = [], seen = new Set();
    for (const e of evs){
      if (/摘录/.test(e.title)) continue;   // 摘录类是外部观点，不产出本人结果数字
      const blob = (e.free["简述"]||"") + " " + Object.values(e.fields).join(" ");
      for (const m of blob.match(re) || []){
        const v = m.replace(/\s+/g, " ").trim();
        if (!seen.has(v)){ seen.add(v); out.push(v); }
        if (out.length >= 3) return out;
      }
    }
    return out;
  };
  const main = grab(RE_MAIN);
  return main.length ? main : grab(RE_ALT);
}

const cut = (s, n) => s.length > n ? s.slice(0, n) + "…" : s;

// ═══════════════ 📅 逐日取数（固定 7 行，缺日记不省略） ═══════════════
const d0 = new Date(+START.slice(0,4), +START.slice(4,6) - 1, +START.slice(6,8));
const key = dt => `${dt.getFullYear()}${String(dt.getMonth()+1).padStart(2,"0")}${String(dt.getDate()).padStart(2,"0")}`;
const byName = new Map();
for (const p of dv.pages(`"${FOLDER}"`).where(p => /^\d{8}$/.test(p.file.name))) byName.set(p.file.name, p);

const rows = [];
const missing = [];
let days = 0, events = 0;
for (let i = 0; i < 7; i++){
  const dt = new Date(d0.getFullYear(), d0.getMonth(), d0.getDate() + i);
  const d = key(dt);
  const label = `${WD[i]}（${d.slice(4,6)}/${d.slice(6,8)}）`;
  const evtLine = (EVT[d] || []).map(([id, txt]) => `**${id}** ${txt}`).join("<br>");
  const p = byName.get(d);
  if (!p){
    missing.push(`${d.slice(4,6)}/${d.slice(6,8)}`);
    rows.push([label, MUTE + "无日记文件</span>", MUTE + "—</span>", MUTE + "—</span>",
               MUTE + "—</span>", evtLine || MUTE + "—</span>"]);
    continue;
  }
  const raw = await dv.io.load(p.file.path);
  days++; events += (raw.match(/^##\s*📌/gm) || []).length;
  const evs = parseDiary(raw);
  if (!evs.length){
    rows.push([label, p.file.link, MUTE + "文件存在，但无 📌 事件</span>", MUTE + "—</span>",
               MUTE + "—</span>", evtLine || MUTE + "—</span>"]);
    continue;
  }
  const inner = evs.map(e => e.fields["洞察"] || e.fields["改进"] || "").filter(Boolean);
  const nums = pickNumbers(evs);
  const tags = [...new Set(evs.flatMap(e => e.themes))].join(" ");
  rows.push([
    label,
    p.file.link,
    inner.length ? inner.map(s => cut(s, 46)).join("<br>") : MUTE + "日记未填「洞察/改进」</span>",
    evs.map((e, k) => (evs.length > 1 ? CIRC[k] + " " : "") + e.title).join("<br>"),
    nums.length ? nums.join(" · ") : MUTE + "无可验证数字</span>",
    (evtLine || MUTE + "未登记事件 ID</span>") + (tags ? `<br><small>${tags}</small>` : "")
  ]);
}

// ═══════════════ 🖼️ 输出 ═══════════════
dv.paragraph(`**${WEEK_NO}** ｜ ${START} ~ ${END} ｜ 有日记 **${days}** 天 ｜ 事件 **${events}** 件 ｜ 缺日记 **${7 - days}** 天${missing.length ? "（" + missing.join("、") + "）" : ""}`);
dv.table(["日期","原始日记","内心","行动","结果","当日关键事件（事件 ID）"], rows);
```
