%%【周报模板 v8】把下面这一整块 `dataviewjs` 粘到 W49.md 里就行（用 Templater 插入也行，里面没有 Templater 标签，会原样落进去）。
它会自动：①认周 ②读本周 7 天日记里的每个 `## 📌` ③**一行画一个事件** ④「日期」用 rowspan 真合并 ⑤内心=该事件的 `- 洞察::`，行动=该事件的 `### 🎯 计划`，结果=该事件的 `### 📝 简述`。
不用再手改行数、不用算 merge 坐标；日记改了什么，切一下阅读模式就更新。%%
```dataviewjs
// ══════════════════════════════════════════════════════════════════════
//  周报渲染器：直接画表（HTML table + rowspan 合并），不依赖 inline 查询
//  想改备注：改下面 NOTE 里 "天:k": "新起/延续/调整/转折/收束/单独"
// ══════════════════════════════════════════════════════════════════════
const NOTE = { /* 例： "0:0": "新起", "0:1": "单独", "0:2": "延续" */ };

const p2 = n => String(n).padStart(2, "0");
const cur = dv.current() || {};
const AF = (typeof app !== "undefined" && app.workspace && app.workspace.getActiveFile) ? app.workspace.getActiveFile() : null;
const C = [];
if (cur.file && cur.file.name) C.push(String(cur.file.name) + " " + String(cur.file.path || ""));
if (AF && AF.basename) C.push(String(AF.basename) + " " + String(AF.path || ""));
const fy = ((C[0] || "").match(/Diary\/(\d{4})\//) || (C[0] || "").match(/(20\d{2})/) || []);
const Y = fy[1] ? +fy[1] : new Date().getFullYear();
const iso = d => {
  const u = new Date(Date.UTC(d.getFullYear(), d.getMonth(), d.getDate()));
  u.setUTCDate(u.getUTCDate() + 4 - (u.getUTCDay() || 7));
  const z = new Date(Date.UTC(u.getUTCFullYear(), 0, 1));
  return Math.ceil(((u - z) / 864e5 + 1) / 7);
};
let WK = 0;
for (const c of C) {
  const mw = c.match(/[Ww](\d{1,2})/);
  if (mw) { WK = +mw[1]; break; }
  const m8 = c.match(/(20\d{2})\D?(\d{2})\D?(\d{2})(?!\d)/);
  if (m8) { const t = iso(new Date(+m8[1], +m8[2] - 1, +m8[3])); if (t >= 1 && t <= 53) { WK = t; break; } }
}
if (!(WK >= 1 && WK <= 53)) WK = iso(new Date());
const jan4 = new Date(Y, 0, 4);
const MON = new Date(jan4.getTime());
MON.setDate(jan4.getDate() - ((jan4.getDay() || 7) - 1) + (WK - 1) * 7);
MON.setHours(0, 0, 0, 0);
const md = d => p2(d.getMonth() + 1) + "/" + p2(d.getDate());
const ymd = d => "" + d.getFullYear() + p2(d.getMonth() + 1) + p2(d.getDate());
const DN = ["周一", "周二", "周三", "周四", "周五", "周六", "周日"];
const dayOf = i => { const d = new Date(MON.getTime()); d.setDate(d.getDate() + i); return d; };
const fileOf = d8 => app.vault.getMarkdownFiles().find(f => f.path.indexOf("Diary/" + Y + "/") === 0 && f.basename.indexOf(d8) === 0);

const clean = s => String(s)
  .replace(/%%[\s\S]*?%%/g, "").replace(/<!--[\s\S]*?-->/g, "").replace(/!\[\[[^\]]*\]\]/g, "")
  .replace(/\[\[([^\]|]*)\|([^\]]*)\]\]/g, "$2").replace(/\[\[([^\]]*)\]\]/g, "$1")
  .replace(/^[ \t]*#{1,6}[ \t]*/gm, "").replace(/\*\*/g, "")
  .split("|").join("／").replace(/\s+/g, " ").trim();
const cut = (s, n) => { const t = clean(s); return t.length > n ? t.slice(0, n) + "…" : t; };
const secOf = (b, name) => {
  const i = b.indexOf("### " + name);
  if (i < 0) return "";
  const rest = b.slice(i + 4 + name.length);
  const j = rest.search(/^#{2,3} /m);
  return j < 0 ? rest : rest.slice(0, j);
};
const parse = raw => raw.replace(/%%[\s\S]*?%%/g, "").split(/^##\s*📌\s*/m).slice(1).map(b => {
  const head = (b.split("\n")[0] || "").trim();
  const score = (b.match(/^[ \t]*-?[ \t]*重要度::[ \t]*(\d)/m) || [])[1] || "";
  const mind = (secOf(b, "🔍 分析").match(/^[ \t]*-?[ \t]*洞察::[ \t]*([^\n]*)$/m) || [])[1] || "";
  const sec = secOf(b, "🎯 计划");
  const w1 = (sec.match(/^[ 	]*[-*]?[ 	]*What::[ 	]*(.*)$/m) || [])[1] || "";
  const w2 = (sec.match(/^[ 	]*[-*]?[ 	]*When::[ 	]*(.*)$/m) || [])[1] || "";
  let act = (w1 || w2) ? [clean(w1), clean(w2)].filter(Boolean).join(" ｜ ") : clean(sec);
  if (/^原文未写，待补 ｜ 原文未写，待补$/.test(act)) act = "原文未写，待补";
  act = act.replace(/\s*｜\s*$/, "").trim();
  const res = clean(secOf(b, "📝 简述")).replace(/^第[-一二三四五六七八九十\d]+件事[：:]\s*/, "").trim();
  return { head, score, mind, act, res };
});
const esc = s => String(s).replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;");
const link = (d8, head, label) =>
  '<a class="internal-link" data-href="' + esc(d8 + "#📌 " + head) + '" href="' + esc(d8 + "#📌 " + head) + '">' + esc(label) + "</a>";

/* ── 读本周 7 天 ── */
const days = [];
for (let i = 0; i < 7; i++) {
  const d8 = ymd(dayOf(i)), f = fileOf(d8);
  let evs = null;
  if (f) { try { evs = parse(await dv.io.load(f.path)); } catch (e) { evs = []; } }
  days.push({ i, d8, has: !!f, evs: evs || [] });
}

/* ── 画表 ── */
const TD = "border:1px solid var(--background-modifier-border);padding:4px 6px;vertical-align:top;";
const TH = TD + "background:var(--background-secondary);text-align:left;font-weight:600;";
const HD = ["日期", "事件（一句话，链到日记那一条）", "内心", "行动", "结果", "备注", "笔记"];
let h = '<table style="border-collapse:collapse;width:100%;font-size:0.92em"><thead><tr>';
for (const x of HD) h += '<th style="' + TH + '">' + x + "</th>";
h += "</tr></thead><tbody>";
let nDay = 0, nEv = 0;
for (const d of days) {
  const evs = d.evs;
  if (d.has) nDay++;
  nEv += evs.length;
  if (!evs.length) {
    h += "<tr>";
    h += '<td style="' + TD + '">' + esc(DN[d.i] + " " + md(dayOf(d.i))) + "</td>";
    h += '<td style="' + TD + '">' + (d.has ? "有日记，但没有 📌 事件" : "暂无日记记录") + "</td>";
    h += '<td style="' + TD + '">—</td><td style="' + TD + '">—</td><td style="' + TD + '">—</td>';
    h += '<td style="' + TD + '">—</td><td style="' + TD + '">—</td></tr>';
    continue;
  }
  evs.forEach((e, k) => {
    h += "<tr>";
    h += k === 0
      ? '<td style="' + TD + '" rowspan="' + evs.length + '">' + esc(DN[d.i] + " " + md(dayOf(d.i))) + "</td>"
      : "";
    h += '<td style="' + TD + '">' + link(d.d8, e.head, cut(e.head, 40)) + "</td>";
    h += '<td style="' + TD + '">' + esc(cut(e.mind, 150) || "—") + "</td>";
    h += '<td style="' + TD + '">' + esc(cut(e.act, 150) || "—") + "</td>";
    h += '<td style="' + TD + '">' + esc(cut(e.res, 150) || "—") + "</td>";
    h += '<td style="' + TD + '">' + esc(NOTE[d.i + ":" + k] || "—") + "</td>";
    h += '<td style="' + TD + '"><a class="internal-link" data-href="' + esc(d.d8) + '" href="' + esc(d.d8) + '">' + esc("[​[" + d.d8 + "]]") + "</a>"
      + " · ★" + esc(e.score || "?") + " · 📌×" + evs.length + "</td>";
    h += "</tr>";
  });
}
h += "</tbody></table>";

const meta = "周期：**" + Y + "-" + md(MON).replace("/", "-") + "（周一）～ " + Y + "-" + md(dayOf(6)).replace("/", "-") + "（周日）**，ISO 第 " + WK + " 周"
  + " ｜ 覆盖：**" + nDay + " / 7 天有日记**，事件 **" + nEv + " 条** ｜ 一行 = 一件事，「日期」列已按事件数合并";

dv.header(1, "W" + WK + " · " + Y + "-" + md(MON).replace("/", "-") + " ~ " + md(dayOf(6)).replace("/", "-"));
dv.paragraph(meta);
dv.container.createEl("div").innerHTML = h;
dv.paragraph("备注列留空时显示「—」，要写得改本块顶部 NOTE（例：`\"0:0\": \"新起\"`）。");
```
