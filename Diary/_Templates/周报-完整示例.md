# 周报模板：隐藏取数引擎 + 纯 Markdown 表格

使用方法：在每篇周报中先插入一次下方取数引擎，再插入“纯 Markdown 表格”部分。实际使用时，建议分别将两个文件设为 Templater 模板：引擎只需插入一次；需要生成表格时，只插入纯 Markdown 表格文件。

---

将下面整个代码块放在每篇周报文件的 Markdown 表格之前（同一篇周报里放一次即可）。
该代码块只定义 WRF 取数函数并隐藏自身，不负责生成表格。

```dataviewjs
// 周报取数引擎：供普通 Markdown 表格中的 $= WRF(dv).xxx(i) 调用
// 每天一行；同一天有多个事件时，事件/洞察/行动/结果在对应单元格内汇总。

const CACHE = globalThis.__WRF_DAILY_CACHE || (globalThis.__WRF_DAILY_CACHE = new Map());

globalThis.WRF = function (dv) {
  const current = dv.current() || {};
  const active = app.workspace.getActiveFile();
  const clues = [
    current.file ? `${current.file.name || ""} ${current.file.path || ""}` : "",
    active ? `${active.basename || ""} ${active.path || ""}` : ""
  ].join(" ");

  const p2 = n => String(n).padStart(2, "0");
  const isoParts = date => {
    const d = new Date(Date.UTC(date.getFullYear(), date.getMonth(), date.getDate()));
    d.setUTCDate(d.getUTCDate() + 4 - (d.getUTCDay() || 7));
    const isoYear = d.getUTCFullYear();
    const jan1 = new Date(Date.UTC(isoYear, 0, 1));
    const week = Math.ceil((((d - jan1) / 86400000) + 1) / 7);
    return { year: isoYear, week };
  };

  let year = 0;
  const pathYear = clues.match(/Diary\/(20\d{2})\//);
  const anyYear = clues.match(/(20\d{2})/);
  if (pathYear) year = Number(pathYear[1]);
  else if (anyYear) year = Number(anyYear[1]);

  let week = 0;
  const weekMatch = clues.match(/(?:^|[^A-Za-z0-9])[Ww](\d{1,2})(?:\b|\.md)/);
  const dateMatch = clues.match(/(20\d{2})\D?(\d{2})\D?(\d{2})(?!\d)/);
  if (weekMatch) week = Number(weekMatch[1]);
  else if (dateMatch) {
    const date = new Date(Number(dateMatch[1]), Number(dateMatch[2]) - 1, Number(dateMatch[3]));
    const parts = isoParts(date);
    week = parts.week;
    if (!pathYear && !anyYear) year = parts.year;
  }

  if (!year) year = isoParts(new Date()).year;
  if (!(week >= 1 && week <= 53)) week = isoParts(new Date()).week;

  // ISO 周一：ISO 年 1 月 4 日所在周为第 1 周。
  const jan4 = new Date(year, 0, 4);
  const monday = new Date(jan4);
  monday.setDate(jan4.getDate() - ((jan4.getDay() || 7) - 1) + (week - 1) * 7);
  monday.setHours(0, 0, 0, 0);

  const dayOf = i => {
    const d = new Date(monday);
    d.setDate(d.getDate() + i);
    return d;
  };
  const ymd = d => `${d.getFullYear()}${p2(d.getMonth() + 1)}${p2(d.getDate())}`;
  const md = d => `${p2(d.getMonth() + 1)}/${p2(d.getDate())}`;
  const dayNames = ["周一", "周二", "周三", "周四", "周五", "周六", "周日"];

  const fileOf = d => {
    const folder = `Diary/${d.getFullYear()}/`;
    const prefix = ymd(d);
    return app.vault.getMarkdownFiles().find(f => f.path.startsWith(folder) && f.basename.startsWith(prefix));
  };

  const clean = value => String(value ?? "")
    .replace(/%%[\s\S]*?%%/g, "")
    .replace(/<!--[\s\S]*?-->/g, "")
    .replace(/!\[\[[^\]]*\]\]/g, "")
    .replace(/\[\[([^\]|]*)\|([^\]]*)\]\]/g, "$2")
    .replace(/\[\[([^\]]*)\]\]/g, "$1")
    .replace(/^[ \t]*#{1,6}[ \t]*/gm, "")
    .replace(/\*\*/g, "")
    .split("|").join("／")
    .replace(/\s+/g, " ")
    .trim();

  const cut = (value, max = 150) => {
    const text = clean(value);
    return text.length > max ? text.slice(0, max) + "…" : text;
  };

  // 三级标题段落，到下一个二级/三级标题结束；#### 不作为段落边界。
  const sectionOf = (block, heading) => {
    const marker = `### ${heading}`;
    const at = block.indexOf(marker);
    if (at < 0) return "";
    const rest = block.slice(at + marker.length);
    const next = rest.search(/^#{2,3}\s/m);
    return next < 0 ? rest : rest.slice(0, next);
  };

  const parse = raw => String(raw || "")
    .replace(/%%[\s\S]*?%%/g, "")
    .split(/^##\s*📌\s*/m)
    .slice(1)
    .map(block => {
      const head = (block.split(/\r?\n/)[0] || "").trim();
      const score = (block.match(/^[ \t]*-?[ \t]*重要度::[ \t]*(\d+)/m) || [])[1] || "";
      const analysis = sectionOf(block, "🔍 分析");
      const mind = (analysis.match(/^[ \t]*-?[ \t]*洞察::[ \t]*([^\r\n]*)$/m) || [])[1] || "";
      const plan = sectionOf(block, "🎯 计划");
      const what = (plan.match(/^[ \t]*[-*]?[ \t]*What::[ \t]*([^\r\n]*)$/mi) || [])[1] || "";
      const when = (plan.match(/^[ \t]*[-*]?[ \t]*When::[ \t]*([^\r\n]*)$/mi) || [])[1] || "";
      let result = sectionOf(block, "📝 简述");
      result = clean(result).replace(/^第[-一二三四五六七八九十\d]+件事[：:]\s*/, "");
      return {
        head,
        score,
        mind: clean(mind),
        act: [clean(what), clean(when)].filter(Boolean).join(" ｜ "),
        result
      };
    });

  // 同一天所有单元格共享一个读取 Promise，避免重复读同一份日记。
  const load = d => {
    const file = fileOf(d);
    if (!file) return Promise.resolve(null);
    const key = `${file.path}:${file.stat?.mtime || 0}`;
    if (!CACHE.has(key)) {
      CACHE.set(key, dv.io.load(file.path).then(parse).catch(error => {
        console.error("周报读取日记失败：", file.path, error);
        CACHE.delete(key);
        return [];
      }));
    }
    return CACHE.get(key);
  };

  const dailyText = async (i, field) => {
    const d = dayOf(i);
    const events = await load(d);
    if (events === null || !events.length) return "—";
    const available = events
      .map((event, index) => ({ event, index, value: event[field] || "" }))
      .filter(item => item.value);
    if (!available.length) return "—";
    const joined = available.map(({ event, index, value }) =>
      events.length > 1 ? `${index + 1}. ${event.head}：${value}` : value
    ).join("；");
    return cut(joined, 150) || "—";
  };

  return {
    week: () => week,
    head: () => `W${week} · ${monday.getFullYear()}-${md(monday).replace("/", "-")} ~ ${dayOf(6).getFullYear()}-${md(dayOf(6)).replace("/", "-")}`,
    span: () => `${monday.getFullYear()}-${md(monday).replace("/", "-")}（周一）～ ${dayOf(6).getFullYear()}-${md(dayOf(6)).replace("/", "-")}（周日）`,
    cover: async () => {
      let diaryDays = 0;
      let eventCount = 0;
      for (let i = 0; i < 7; i++) {
        const events = await load(dayOf(i));
        if (events !== null) diaryDays++;
        eventCount += events ? events.length : 0;
      }
      return `${diaryDays} / 7 天有日记｜事件 ${eventCount} 条`;
    },
    d: i => `${dayNames[i]} ${md(dayOf(i))}`,
    ev: async i => {
      const d = dayOf(i);
      const events = await load(d);
      if (events === null) return "暂无日记记录";
      if (!events.length) return "有日记，但没有 📌 事件";
      const links = events.map(event => dv.sectionLink(
        `Diary/${d.getFullYear()}/${ymd(d)}`,
        `📌 ${event.head}`,
        false,
        cut(event.head, 60)
      ));
      return links.length === 1 ? links[0] : links;
    },
    mind: i => dailyText(i, "mind"),
    act: i => dailyText(i, "act"),
    res: i => dailyText(i, "result"),
    notes: async i => {
      const d = dayOf(i);
      const events = await load(d);
      if (events === null) return "—";
      const scores = [...new Set(events.map(e => e.score).filter(Boolean))];
      const scoreText = scores.length ? scores.join("/") : "?";
      const label = `${ymd(d)} · ★${scoreText} · 📌×${events.length}`;
      return dv.fileLink(`Diary/${d.getFullYear()}/${ymd(d)}`, false, label);
    }
  };
};

// 隐藏引擎代码块本身，只保留其供 inline 查询调用的全局函数。
this.container.style.display = "none";
```
