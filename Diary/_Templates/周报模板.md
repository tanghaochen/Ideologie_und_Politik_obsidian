# W{{周号}} · {{起}} ~ {{止}}（ISO 第 {{周号}} 周）

> 周期：**{{起}}（周一）～ {{止}}（周日）**，ISO 第 {{周号}} 周
>
> 覆盖：{{d}}/7 天有日记 ｜ 事件 {{n}} 条 ｜ 事件链 {{m}} 条
>
> 本页两张表分工固定：**一、日记索引**由 DataviewJS 读日记模板字段自动生成（只读、不合并、不排版）；**二、本周事件总表**是原生 Markdown，承载人工判读、事件 ID 与合并单元格。两表不一致时以日记原文为准。

## 一、日记索引（DataviewJS · 只读，勿手改）

> 逐条索引本周 7 个日期文件的日记模板字段：📌 标题 / 重要度 / 主题 / 🔍 洞察·改进 / 🎯 计划 / 原文数字 / 字段完整度。
> 它只回答两件事：**有没有漏事件**、**模板哪些字段没填**。单元格里的 `⟨分析…⟩` 表示该字段为空、用「🔍 分析」正文兜底，不是日记原字段。
> 日期与事件 ID 自动推算，**换周只需改文件名**；`EVT` 里可选手填事件 ID。

```dataviewjs
// ═══════════════ ⚙️ 配置区（一般不用改） ═══════════════
const FOLDER = "Diary/2026";   // 日记目录（Vault 内相对路径，不要写 E:\…）
const START  = "";             // 留空 = 按本文件名 W{ISO周} 自动算周一；要手填时写 "20260928"
const END    = "";             // 留空 = 周一 + 6 天
const EVT    = {};             // 事件 ID（可选）：{ "20260929": [["event-2026-004", "一句话"]] }

// ═══════════════ 🔧 工具 ═══════════════
const WD = ["周一","周二","周三","周四","周五","周六","周日"];
const CIRC = ["①","②","③","④","⑤","⑥","⑦","⑧"];
const MUTE = '<span style="opacity:.55">';
const OK = "✓", NO = "✗";
const pad = n => String(n).padStart(2, "0");
const keyOf = d => `${d.getFullYear()}${pad(d.getMonth()+1)}${pad(d.getDate())}`;
const parseKey = s => new Date(+s.slice(0,4), +s.slice(4,6) - 1, +s.slice(6,8));
const flat = s => String(s ?? "").replace(/\s+/g, " ").trim();
const cut = (s, n) => { const t = flat(s); return t.length > n ? t.slice(0, n) + "…" : t; };
const firstClause = (s, n) => { const t = flat(s); return cut(t.split(/[。；;！？!?]/)[0] || t, n); };
const PLAN_KEYS = ["What","Who","When","Where","How","预算","预期"];
const RE_NUM = /(\d+(?:\.\d+)?\s*个?\s*(?:小时|分钟|元|块|次|条|步|篇|天|人)|\d+\s*\/\s*100)/g;
const RE_ALT = /([一二两三四五六七八九十]+个?(?:小时|分钟)|\d{1,2}\s*Uhr|(?:eine|einer|zwei|drei|vier|fünf|sechs|sieben|acht|neun|zehn)\s+(?:Stunden?|Folgen?)|(?:über\s+)?eine\s+halbe\s+Stunde)/gi;

// ISO 周 → 该周周一（第 1 周 = 含 1 月 4 日的那一周，周一起始）
function isoMonday(year, week){
  const jan4 = new Date(year, 0, 4);
  const mon1 = new Date(year, 0, 4 - ((jan4.getDay() + 6) % 7));
  return new Date(mon1.getFullYear(), mon1.getMonth(), mon1.getDate() + (week - 1) * 7);
}

// 只读解析一篇日记：事件分隔符 "## 📌"，字段 "key:: value"，章节 "### xxx"
// %% 提示语 %%、<!-- -->、![[嵌入]]、子标题本身都不进入正文
function parseDiary(raw){
  const out = [];
  const body = String(raw ?? "").replace(/%%[\s\S]*?%%/g, "").replace(/<!--[\s\S]*?-->/g, "");
  for (const b of body.split(/^##\s*📌\s*/m).slice(1)){
    const lines = b.split("\n");
    const ev = { title:(lines[0]||"").trim(), rating:0, themes:[], persons:[], fields:{}, plan:{}, free:{} };
    let sec = "";
    for (const line of lines.slice(1)){
      if (/^\s*!\[\[/.test(line)) continue;
      const h = line.match(/^###\s+([^#\s][^\n]*)$/);
      if (h){ sec = h[1].replace(/[^\u4e00-\u9fa5A-Za-z]/g, ""); continue; }
      if (/^#{1,6}\s/.test(line)) continue;
      const f = line.match(/^\s*-?\s*([\w\u4e00-\u9fa5]+)::\s*(.*)$/);
      if (f){
        const k = f[1], v = (f[2]||"").trim();
        if (k === "重要度"){ ev.rating = Math.min(5, parseInt(v) || 0); continue; }
        if (k === "主题"){ ev.themes = v.split(/[\s,，]+/).filter(t => t.startsWith("#")); continue; }
        if (k === "人物"){ if (v) ev.persons.push(v); continue; }
        if (!v) continue;
        if (PLAN_KEYS.includes(k)) ev.plan[k] = v;
        else ev.fields[k] = ev.fields[k] ? ev.fields[k] + "；" + v : v;
        continue;
      }
      const t = line.trim();
      if (sec && t && !/^[-*|`>]/.test(t)) ev.free[sec] = (ev.free[sec] || "") + t + " ";
    }
    out.push(ev);
  }
  return out;
}

function pickNums(...texts){
  const out = [], seen = new Set();
  const scan = re => {
    for (const t of texts){
      for (const m of String(t || "").match(re) || []){
        const v = m.replace(/\s+/g, " ").trim();
        if (!seen.has(v)){ seen.add(v); out.push(v); }
        if (out.length >= 4) return true;
      }
    }
    return false;
  };
  if (!scan(RE_NUM)) scan(RE_ALT);
  return out;
}

// ═══════════════ 🧭 周期：文件名 W39 → ISO 第 39 周的周一 ═══════════════
const self = dv.current();
const dirYear = (self.file.path.match(/\/(\d{4})\//) || [])[1];
const wNo = (self.file.name.match(/W(\d{1,2})/i) || [])[1];
const year = dirYear ? +dirYear : new Date().getFullYear();
const d0 = START ? parseKey(START) : (wNo ? isoMonday(year, +wNo) : null);

async function render(d0){
  const d1 = END ? parseKey(END) : new Date(d0.getFullYear(), d0.getMonth(), d0.getDate() + 6);
  const byName = new Map();
  for (const p of dv.pages(`"${FOLDER}"`).where(p => /^\d{8}$/.test(p.file.name))) byName.set(p.file.name, p);

  const rows = [], missing = [];
  let days = 0, evCount = 0, hot = 0, files = 0;

  for (let i = 0; i < 7; i++){
    const dt = new Date(d0.getFullYear(), d0.getMonth(), d0.getDate() + i);
    const d = keyOf(dt), md = `${d.slice(4,6)}/${d.slice(6,8)}`;
    const label = `${WD[i]} ${md}`;
    const p = byName.get(d);
    if (!p){
      missing.push(md);
      rows.push([label, MUTE + "无日记文件</span>", MUTE + "—</span>", MUTE + "—</span>",
                 MUTE + "—</span>", MUTE + "—</span>", MUTE + "—</span>"]);
      continue;
    }
    files++;
    const raw = await dv.io.load(p.file.path);
    const evs = parseDiary(raw);
    if (!evs.length){
      rows.push([label, p.file.link, MUTE + "文件存在，无 📌 事件</span>", MUTE + "—</span>",
                 MUTE + "—</span>", MUTE + "—</span>", MUTE + "空文件 / 未写事件</span>"]);
      continue;
    }
    days++;
    evs.forEach((ev, k) => {
      evCount++;
      if (ev.rating >= 5) hot++;
      const ids = (EVT[d] || []).map(([id]) => id);
      const filled = { 简述: !!flat(ev.free["简述"]), 洞察: !!ev.fields["洞察"], 改进: !!ev.fields["改进"],
                       计划: !!flat(ev.free["计划"]) || Object.keys(ev.plan).length > 0, 推演: !!flat(ev.free["推演"]) };
      let nums = pickNums(ev.free["简述"], ev.fields["洞察"], ev.fields["改进"], ev.free["计划"], ev.free["推演"]);
      let numFromAnalysis = false;
      if (!nums.length){ nums = pickNums(ev.free["分析"]); numFromAnalysis = nums.length > 0; }

      const inner = ev.fields["洞察"] || ev.fields["改进"];
      const innerCell = inner ? cut(inner, 76)
        : (flat(ev.free["分析"]) ? MUTE + "⟨分析首句⟩</span> " + firstClause(ev.free["分析"], 64)
                                 : MUTE + "未填「洞察 / 改进」</span>");
      const planText = flat(ev.free["计划"]) || Object.entries(ev.plan).map(([a, b]) => `${a}：${b}`).join("；");
      const planCell = planText ? cut(planText, 76)
        : (flat(ev.free["分析"]) ? MUTE + "⟨分析⟩</span> " + firstClause(ev.free["分析"], 64)
                                 : MUTE + "计划栏为空</span>");

      rows.push([
        label,
        p.file.link,
        `${evs.length > 1 ? CIRC[k] + " " : ""}**${ev.title}**` + (ev.rating ? ` <small>★${ev.rating}</small>` : "") + (ev.themes.length ? `<br><small>${ev.themes.join(" ")}</small>` : "") + (ev.persons.length ? `<br><small>👤 ${ev.persons.map(p => cut(p, 24)).join(" / ")}</small>` : ""),
        innerCell,
        planCell,
        nums.length ? nums.join(" · ") + (numFromAnalysis ? ` <small>${MUTE}（取自分析）</span></small>` : "") : MUTE + "原文无可核对数字</span>",
        (ids.length ? ids.join("<br>") : MUTE + "未登记 ID</span>") + `<br><small>${Object.entries(filled).map(([k, v]) => `${k[0]} ${v ? OK : NO}`).join(" ")}</small>`
      ]);
    });
  }

  dv.paragraph(`**${keyOf(d0)} ~ ${keyOf(d1)}** ｜ 有日记文件 **${files}** 天（其中有事件 **${days}** 天）｜ 事件 **${evCount}** 件${hot ? `（重要度 5：**${hot}** 件）` : ""} ｜ 缺日记 **${7 - files}** 天${missing.length ? "（" + missing.join("、") + "）" : ""}`);
  dv.table(["日期","日记","事件（📌 · ★ · #主题）","🔍 洞察 / 改进","🎯 计划","原文数字","当日事件 ID · 字段"], rows);
}

if (!d0) dv.paragraph("⚠️ 无法判断周期：文件名里没有 `W{周号}`，请在配置区填写 `START`。");
else await render(d0);
```

【模板使用说明 · 新建报告后请删除本节】

1. 两张表的分工不能混：
   - 第一节 `dataviewjs` **自动**读日记模板（📌 标题 / 重要度 / 主题 / 洞察 / 改进 / 计划 / 推演），只做索引与漏项核对；`dv.table()` 输出 HTML，**不合并、不手改**，内容错了去改日记。
   - 第二节是 **原生 Markdown**，写人工判读、事件 ID，`merge` 就写在它下面的注释里。
2. DataviewJS 的配置只有 4 个：`FOLDER`、`START`、`END`、`EVT`。正常情况下**一个都不用改**：文件名 `W39.md` 会自动解析成 ISO 第 39 周，「Diary/2026」会自动取年份。只有文件名不规范时才填 `START`。
3. `EVT` 只影响「当日事件 ID」列，可留空。它按**日期**登记，所以同一天有多个事件时会共用同一批 ID；跨周期事件沿用同一个 `event-2026-NNN`，**不要重新编号**。
4. **行数一变就必须重算表下的 merge 坐标**：行 1 是表头，`---` 分隔行不计行号。
   当前模板 8 个数据行（第 2–9 行）：`A2:A3` = 周一的两行，`G2:G3` = 周一的两条笔记链接。
   加行/删行后整条注释重写，不要手改数字；单格范围（如 `A5:A5`）会被插件丢弃，所以只出现一行的那天不要写 merge。
5. 字段映射（避免瞎填）：
   - 内心 ← 日记「洞察」+「改进」+ 简述里的感受、价值判断
   - 行动 ← 日记「🎯计划」正文 + 简述里实际做的事（两者要分开写时用 `<br>`）
   - 结果 ← 简述或后续日记里的反馈；没有反馈就写「尚无结果」，不许编
6. 「事件（一句话）」≤30 字，说清"做了什么"而不是"想了什么"；同一天多事件就加行并合并日期与笔记，不要挤在一格里。
7. 某天没有日记：保留一行，事件写「暂无日记记录」，内心/行动/结果写「暂无记录」，备注写「—」，笔记写「—」（不要留空链接）。
8. 格子里不要用 `##` 标题、不要用 `-` 列表符号；需要换行用 `<br>`；空值统一写「暂无记录」。
9. 表下那句"空白格 = 与上一行同组"不要删——没装插件时靠它读表。
