const p2 = n => String(n).padStart(2, "0");
const cur = dv.current() || {};
const AF = (typeof app !== "undefined" && app.workspace && app.workspace.getActiveFile) ? app.workspace.getActiveFile() : null;

const SRC = cur.file ? String(cur.file.path || "") : "";
const isoParts = d0 => {
  const d = new Date(Date.UTC(d0.getFullYear(), d0.getMonth(), d0.getDate()));
  d.setUTCDate(d.getUTCDate() + 4 - (d.getUTCDay() || 7));
  const y = d.getUTCFullYear(), j1 = new Date(Date.UTC(y, 0, 1));
  return { year: y, week: Math.ceil((((d - j1) / 864e5) + 1) / 7) };
};

const py = clues.match(/Diary\/(20\d{2})\//), ay = clues.match(/(20\d{2})/);
if (py) Y = Number(py[1]); else if (ay) Y = Number(ay[1]);

const mw = clues.match(/(?:^|[^A-Za-z0-9])[Ww](\d{1,2})(?:\b|\.md)/);
const m8 = clues.match(/(20\d{2})\D?(\d{2})\D?(\d{2})(?!\d)/);
if (mw) WK = Number(mw[1]);
else if (m8) { const p = isoParts(new Date(Number(m8[1]), Number(m8[2]) - 1, Number(m8[3]))); WK = p.week; if (!Y) Y = p.year; }
if (!Y) Y = isoParts(new Date()).year;
if (!(WK >= 1 && WK <= 53)) WK = isoParts(new Date()).week;

  .replace(/%%[\s\S]*?%%/g, "").replace(/<!--[\s\S]*?-->/g, "").replace(/!\[\[[^\]]*\]\]/g, "")
  .replace(/\[\[([^\]|]*)\|([^\]]*)\]\]/g, "$2").replace(/\[\[([^\]]*)\]\]/g, "$1")

  const at = b.indexOf("### " + name);
  if (at < 0) return "";
  const rest = b.slice(at + 4 + name.length);
  const next = rest.search(/^#{2,3}\s/m);
  return next < 0 ? rest : rest.slice(0, next);
};
const parse = raw => String(raw || "").replace(/%%[\s\S]*?%%/g, "").split(/^##\s*📌\s*/m).slice(1).map(b => {
  const head = (b.split(/\r?\n/)[0] || "").trim();
  const score = (b.match(/^[ \t]*-?[ \t]*重要度::[ \t]*(\d+)/m) || [])[1] || "";
  const analysis = secOf(b, "🔍 分析");

  const plan = secOf(b, "🎯 计划");

  return { head, score, mind: clean(mind), act: [clean(what), clean(when)].filter(Boolean).join(" ｜ "), res };
});

globalThis.WRF = function (dv) {
  const here = (dv.current() || {}).file || null;
  const MISMATCH = !!(here && SRC && String(here.path || "") !== SRC);
  const WARN = "⚠ 本文件没有插「周报-取数引擎」，这是别的周报的数据；请先在本文件顶部插入引擎";
  const day = i => DAYS[i] || { d: dayOf(i), has: false, evs: [] };
  const joined = (i, field) => {
    const d = day(i);
    if (!d.evs.length) return "—";
    const parts = d.evs.map((e, k) => (e[field] || "").trim()).filter(Boolean);
    if (!parts.length) return "—";
    if (d.evs.length === 1) return cut(parts[0]);
    return cut(d.evs.map((e, k) => `${k + 1}. ${cut(e[field] || "—", 40)}`).join("<br>"));
  };
  const API = {
    week: () => WK,
    head: () => `W${WK} · ${MON.getFullYear()}-${md(MON).replace("/", "-")} ~ ${dayOf(6).getFullYear()}-${md(dayOf(6)).replace("/", "-")}`,
    span: () => `${MON.getFullYear()}-${md(MON).replace("/", "-")}（周一）～ ${dayOf(6).getFullYear()}-${md(dayOf(6)).replace("/", "-")}（周日）`,
    cover: () => {
      let n = 0, e = 0;
      for (const d of DAYS) { if (d.has) n++; e += d.evs.length; }
      return `${n} / 7 天有日记｜事件 ${e} 条`;
    },
    d: i => `${DN[i]} ${md(dayOf(i))}`,
    ev: i => {
      const d = day(i);
      if (!d.has) return "暂无日记记录";
      if (!d.evs.length) return "有日记，但没有 📌 事件";
      const links = d.evs.map(e => dv.sectionLink(`Diary/${d.d.getFullYear()}/${ymd(d.d)}`, `📌 ${e.head}`, false, cut(e.head, 60)));
      return links.length === 1 ? links[0] : links;
    },
    mind: i => joined(i, "mind"),
    act: i => joined(i, "act"),
    res: i => joined(i, "res"),
    notes: i => {
      const d = day(i);
      if (!d.has) return "—";
      const scores = [...new Set(d.evs.map(e => e.score).filter(Boolean))];
      return dv.fileLink(`Diary/${d.d.getFullYear()}/${ymd(d.d)}`, false, `${ymd(d.d)} · ★${scores.length ? scores.join("/") : "?"} · 📌×${d.evs.length}`);
    }
  };
  const out = API;
  if (MISMATCH) { for (const k of Object.keys(out)) out[k] = () => WARN; out.__src = SRC; return out; }
  out.__src = SRC;
  return out;
};
