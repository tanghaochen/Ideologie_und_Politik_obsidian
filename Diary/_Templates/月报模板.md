## 二、自动索引（DataviewJS · 只读，勿手改）

> 两张表都由 DataviewJS 现算：**逐周表**核对每周覆盖几天、有几件事件、周报是否存在、模板字段填了几项；**逐事件表**把当月每个 📌 事件按日期铺开，带上洞察 / 改进 / 计划与事件 ID。
> 它不合并、不排版，也不出现在年报嵌入里；内容错了去改日记或周报，不要在这里改。

```dataviewjs
// ═══════════════ ⚙️ 配置区（一般不用改） ═══════════════
const FOLDER = "Diary/2026";   // 日记与周报所在目录（Vault 内相对路径，不要写 E:\…）
const MONTH  = "";             // 留空 = 按本文件名 YYYYMM 自动；要手填时写 "202609"
const EVT    = {};             // 事件 ID（可选）：{ "20260929": [["event-2026-004", "一句话"]] }

// ═══════════════ 🔧 工具 ═══════════════
const MUTE = '<span style="opacity:.55">';
const OK = "✓", NO = "✗";
const pad = n => String(n).padStart(2, "0");
const keyOf = d => `${d.getFullYear()}${pad(d.getMonth()+1)}${pad(d.getDate())}`;
const flat = s => String(s ?? "").replace(/\s+/g, " ").trim();
const cut = (s, n) => { const t = flat(s); return t.length > n ? t.slice(0, n) + "…" : t; };
const PLAN_KEYS = ["What","Who","When","Where","How","预算","预期"];
const RE_NUM = /(\d+(?:\.\d+)?\s*个?\s*(?:小时|分钟|元|块|次|条|步|篇|天|人)|\d+\s*\/\s*100)/g;
const RE_ALT = /([一二两三四五六七八九十]+个?(?:小时|分钟)|\d{1,2}\s*Uhr|(?:eine|einer|zwei|drei|vier|fünf|sechs|sieben|acht|neun|zehn)\s+(?:Stunden?|Folgen?)|(?:über\s+)?eine\s+halbe\s+Stunde)/gi;
const CN = ["第一周","第二周","第三周","第四周","第五周","第六周"];

// ISO 8601 周号（周四归属法）
function isoWeek(dt){
  const d = new Date(dt.getFullYear(), dt.getMonth(), dt.getDate());
  d.setDate(d.getDate() - ((d.getDay() + 6) % 7) + 3);
  const y = d.getFullYear();
  const firstThu = new Date(y, 0, 4);
  firstThu.setDate(4 - ((firstThu.getDay() + 6) % 7) + 3);
  return 1 + Math.round((d - firstThu) / 604800000);
}

function parseDiary(raw){
  const out = [];
  const body = String(raw ?? "").replace(/%%[\s\S]*?%%/g, "").replace(/<!--[\s\S]*?-->/g, "");
  for (const b of body.split(/^##\s*📌\s*/m).slice(1)){
    const lines = b.split("\n");
    const ev = { title:(lines[0]||"").trim(), rating:0, themes:[], persons:[], fields:{}, plan:{}, free:{} };
    let sec = "";
    for (const line of lines.slice(1)){
      if (/^\s*!\[\[/.test(line)) continue;
      const h = line.match(/^###\s+([^#\s][^\n]*)$/);
      if (h){ sec = h[1].replace(/[^\u4e00-\u9fa5A-Za-z]/g, ""); continue; }
      if (/^#{1,6}\s/.test(line)) continue;
      const f = line.match(/^\s*-?\s*([\w\u4e00-\u9fa5]+)::\s*(.*)$/);
      if (f){
        const k = f[1], v = (f[2]||"").trim();
        if (k === "重要度"){ ev.rating = Math.min(5, parseInt(v) || 0); continue; }
        if (k === "主题"){ ev.themes = v.split(/[\s,，]+/).filter(t => t.startsWith("#")); continue; }
        if (k === "人物"){ if (v) ev.persons.push(v); continue; }
        if (!v) continue;
        if (PLAN_KEYS.includes(k)) ev.plan[k] = v;
        else ev.fields[k] = ev.fields[k] ? ev.fields[k] + "；" + v : v;
        continue;
      }
      const t = line.trim();
      if (sec && t && !/^[-*|`>]/.test(t)) ev.free[sec] = (ev.free[sec] || "") + t + " ";
    }
    out.push(ev);
  }
  return out;
}

function pickNums(...texts){
  const out = [], seen = new Set();
  const scan = re => {
    for (const t of texts){
      for (const m of String(t || "").match(re) || []){
        const v = m.replace(/\s+/g, " ").trim();
        if (!seen.has(v)){ seen.add(v); out.push(v); }
        if (out.length >= 3) return true;
      }
    }
    return false;
  };
  if (!scan(RE_NUM)) scan(RE_ALT);
  return out;
}

const filledOf = ev => ({
  简述: !!flat(ev.free["简述"]), 洞察: !!ev.fields["洞察"], 改进: !!ev.fields["改进"],
  计划: !!flat(ev.free["计划"]) || Object.keys(ev.plan).length > 0, 推演: !!flat(ev.free["推演"])
});

// ═══════════════ 🧭 周期：文件名 202609 → 该自然月 ═══════════════
const self = dv.current();
const mStr = MONTH || ((self.file.name.match(/(\d{4})(\d{2})/) || []).slice(1).join(""));
const y = +String(mStr).slice(0, 4), mo = +String(mStr).slice(4, 6);

async function render(){
  const start = new Date(y, mo - 1, 1), end = new Date(y, mo, 0);
  const byName = new Map(), wkLink = new Map();
  for (const p of dv.pages(`"${FOLDER}"`)){
    if (/^\d{8}$/.test(p.file.name)) byName.set(p.file.name, p);
    if (/^W\d{1,2}$/.test(p.file.name)) wkLink.set(p.file.name, p.file.link);
  }

  // 按 ISO 周切段（只保留落在本月内的日期）
  const weeks = [];
  for (let day = 1; day <= end.getDate(); day++){
    const dt = new Date(y, mo - 1, day), k = keyOf(dt), iso = isoWeek(dt);
    let w = weeks[weeks.length - 1];
    if (!w || w.iso !== iso){ w = { no: weeks.length + 1, iso, from: k, to: k, keys: [] }; weeks.push(w); }
    w.to = k; w.keys.push(k);
  }

  const wRows = [], eRows = [];
  const total = { files: 0, days: 0, ev: 0, hot: 0 };
  const fill = { 简述: 0, 洞察: 0, 改进: 0, 计划: 0, 推演: 0 };

  for (const w of weeks){
    const covered = w.keys.map(k => `${k.slice(4,6)}/${k.slice(6,8)}`);
    const have = [];
    let evCount = 0, hotCount = 0;
    const f = { 简述: 0, 洞察: 0, 改进: 0, 计划: 0, 推演: 0 };

    for (const k of w.keys){
      const p = byName.get(k);
      if (!p) continue;
      total.files++; have.push(`${k.slice(4,6)}/${k.slice(6,8)}`);
      const raw = await dv.io.load(p.file.path);
      const evs = parseDiary(raw);
      if (!evs.length) continue;
      total.days++;
      evCount += evs.length;
      for (const ev of evs){
        if (ev.rating >= 5) hotCount++;
        const fl = filledOf(ev);
        for (const key of Object.keys(f)) if (fl[key]) f[key]++;
        let nums = pickNums(ev.free["简述"], ev.fields["洞察"], ev.fields["改进"], ev.free["计划"], ev.free["推演"]);
        if (!nums.length) nums = pickNums(ev.free["分析"]);
        const inner = ev.fields["洞察"] || ev.fields["改进"];
        const planText = flat(ev.free["计划"]) || Object.entries(ev.plan).map(([a, b]) => `${a}：${b}`).join("；");
        eRows.push([
          `${k.slice(4,6)}/${k.slice(6,8)}`,
          `**${ev.title}**` + (ev.rating ? ` <small>★${ev.rating}</small>` : "") + (ev.themes.length ? `<br><small>${ev.themes.join(" ")}</small>` : "") + (ev.persons.length ? `<br><small>👤 ${ev.persons.map(p => cut(p, 24)).join(" / ")}</small>` : ""),
          inner ? cut(inner, 64) : MUTE + "未填「洞察 / 改进」</span>",
          planText ? cut(planText, 64) : MUTE + "计划栏为空</span>",
          nums.length ? nums.join(" · ") : MUTE + "无</span>",
          (EVT[k] || []).length ? EVT[k].map(x => x[0]).join("<br>") : MUTE + "未登记</span>",
          p.file.link
        ]);
      }
    }

    total.ev += evCount; total.hot += hotCount;
    for (const key of Object.keys(fill)) fill[key] += f[key];
    const found = wkLink.get(`W${w.iso}`);
    wRows.push([
      CN[w.no - 1] || `第${w.no}周`,
      `W${w.iso}`,
      covered.length > 1 ? `${covered[0]}–${covered[covered.length - 1]}` : covered[0],
      have.length ? have.join("、") : MUTE + "无</span>",
      String(evCount),
      String(hotCount || 0),
      `简 ${f["简述"]} · 洞 ${f["洞察"]} · 改 ${f["改进"]} · 计 ${f["计划"]} · 推 ${f["推演"]}`,
      found || MUTE + "无周报</span>"
    ]);
  }

  const totalDays = end.getDate();
  dv.paragraph(
    `**${y}${pad(mo)}** ｜ 本月 **${totalDays}** 天 ｜ 有日记文件 **${total.files}** 天（其中写了事件 **${total.days}** 天）｜ 事件 **${total.ev}** 件（★5：**${total.hot}** 件）｜ 缺日记 **${totalDays - total.files}** 天`
  );
  dv.paragraph(
    `**日记模板执行度**（${total.ev} 个事件里填了几项）：简述 **${fill["简述"]}/${total.ev}** · 洞察 **${fill["洞察"]}/${total.ev}** · 改进 **${fill["改进"]}/${total.ev}** · 计划 **${fill["计划"]}/${total.ev}** · 推演 **${fill["推演"]}/${total.ev}**`
  );
  dv.table(["周","ISO","本月覆盖","有日记","事件","★5","字段填充（简述/洞察/改进/计划/推演）","周报"], wRows);
  dv.table(["日期","事件（📌 · ★ · #主题）","🔍 洞察 / 改进","🎯 计划","原文数字","当日事件 ID","笔记"], eRows);
}

if (!/^\d{4}\d{2}$/.test(String(mStr))) dv.paragraph("⚠️ 无法判断月份：文件名不是 `YYYYMM`，请在配置区填写 `MONTH`。");
else await render();
```

【模板使用说明 · 新建报告后请删除本节】

1. 三节分工不能混：
   - `## 月报说明` 下的三段式主表是 **原生 Markdown**，可编辑、可合并、可被年报 Excalidraw 嵌入；`merge` 注释紧贴它最后一行。
   - `## 二、自动索引` 里的两张表是 `dataviewjs` + `dv.table()` 动态渲染，**不合并、不手改**，只做逐周/逐事件核对与模板执行度统计。
   - 两者不一致时以日记为准：补第一节，不要改代码去迁就结论。
2. DataviewJS 只认 3 个配置：`FOLDER`、`MONTH`、`EVT`。正常情况下**一个都不用改**：文件名 `202609.md` 自动解析成 2026 年 9 月，周次按 ISO 8601 从日期现算。
   `EVT` 按**日期**登记事件 ID，同一天多个事件共用同一批 ID；只影响「当日事件 ID」列。
3. 三段结构固定：`简单概述`（整月只看这几行）→ `坚持的` → `未来要解决的`。类别写纯文字，**不要写 `## 简单概述`**（在格子里只会显示成字面的 "##"）。
4. **行数一变就必须重算 merge 坐标**：行 1 是表头，`---` 分隔行不计行号。当前模板 12 个数据行（第 2–13 行）：
   - `A2:A6` = 简单概述五周，`A7:A10` = 坚持的四行，`A11:A13` = 未来要解决的三行
   - `B7:B8`、`B9:B10`、`B12:B13` = 类别内部按周切分；只出现一次周的格子（如 `B11`）**不要**写 merge
   - 概述块按当月实际周数写满（常见 4–5 行，最多 6 行），事件类每类尽量 ≤4 行；不要为了凑行数编周次
5. 常见错误：整列一条通到底（如 `A2:A13`）会把三类并成一个格、文字按行堆在一起；正确做法是每类各自一个区间。
6. 「事件（一句话）」≤30 字，写"做了什么 + 变成什么"；跨周事件在备注里写清「延续自第 N 周」「→ 第 N 周降档」，这就是跨周连接，不必再单开一张事件链表。
7. 笔记列给**能下钻到细节的那一层**：简单概述给 `[[W…]]`（周报），坚持的 / 未来要解决的给 `[[YYYYMMDD]]`（当天日记）；必要用 ` · ` 连两个链接。
8. 某周没有周报：保留该周行，事件写「本周无周报」，结果写「暂无记录」，笔记写「—」。
9. 格子里不要用 `##` 标题、不要用 `-` 列表符号；换行用 `<br>`；空值统一写「暂无记录」。
10. `## 月报说明` 这个标题**不要改**——年报 Excalidraw 按它嵌入本表；表下那句"空白格 = 与上一行同组"也不要删。
