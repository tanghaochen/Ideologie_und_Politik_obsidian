// ═══════════════ 🔧 工具 ═══════════════
const MUTE = '<span style="opacity:.55">';
const pad = n => String(n).padStart(2, "0");
const keyOf = d => `${d.getFullYear()}${pad(d.getMonth()+1)}${pad(d.getDate())}`;
const flat = s => String(s ?? "").replace(/\s+/g, " ").trim();
const cut = (s, n) => { const t = flat(s); return t.length > n ? t.slice(0, n) + "…" : t; };
const PLAN_KEYS = ["What","Who","When","Where","How","预算","预期"];
const RE_NUM = /(\d+(?:\.\d+)?\s*个?\s*(?:小时|分钟|元|块|次|条|步|篇|天|人)|\d+\s*\/\s*100)/g;
const RE_ALT = /([一二两三四五六七八九十]+个?(?:小时|分钟)|\d{1,2}\s*Uhr|(?:eine|einer|zwei|drei|vier|fünf|sechs|sieben|acht|neun|zehn)\s+(?:Stunden?|Folgen?)|(?:über\s+)?eine\s+halbe\s+Stunde)/gi;

// ISO 8601 周号（周四归属法）
function isoWeek(dt){
  const d = new Date(dt.getFullYear(), dt.getMonth(), dt.getDate());
  d.setDate(d.getDate() - ((d.getDay() + 6) % 7) + 3);
  const y = d.getFullYear();
  const firstThu = new Date(y, 0, 4);
  firstThu.setDate(4 - ((firstThu.getDay() + 6) % 7) + 3);
  return 1 + Math.round((d - firstThu) / 604800000);
}

function parseDiary(raw){
  const out = [];
  const body = String(raw ?? "").replace(/%%[\s\S]*?%%/g, "").replace(/<!--[\s\S]*?-->/g, "");
  for (const b of body.split(/^##\s*📌\s*/m).slice(1)){
    const lines = b.split("\n");
    const ev = { title:(lines[0]||"").trim(), rating:0, themes:[], persons:[], fields:{}, plan:{}, free:{} };
    let sec = "";
    for (const line of lines.slice(1)){
      if (/^\s*!\[\[/.test(line)) continue;
      const h = line.match(/^###\s+([^#\s][^\n]*)$/);
      if (h){ sec = h[1].replace(/[^\u4e00-\u9fa5A-Za-z]/g, ""); continue; }
      if (/^#{1,6}\s/.test(line)) continue;
      const f = line.match(/^\s*-?\s*([\w\u4e00-\u9fa5]+)::\s*(.*)$/);
      if (f){
        const k = f[1], v = (f[2]||"").trim();
        if (k === "重要度"){ ev.rating = Math.min(5, parseInt(v) || 0); continue; }
        if (k === "主题"){ ev.themes = v.split(/[\s,，]+/).filter(t => t.startsWith("#")); continue; }
        if (k === "人物"){ if (v) ev.persons.push(v); continue; }
        if (!v) continue;
        if (PLAN_KEYS.includes(k)) ev.plan[k] = v;
        else ev.fields[k] = ev.fields[k] ? ev.fields[k] + "；" + v : v;
        continue;
      }
      const t = line.trim();
      if (sec && t && !/^[-*|`>]/.test(t)) ev.free[sec] = (ev.free[sec] || "") + t + " ";
    }
    out.push(ev);
  }
  return out;
}

function pickNums(...texts){
  const out = [], seen = new Set();
  const scan = re => {
    for (const t of texts){
      for (const m of String(t || "").match(re) || []){
        const v = m.replace(/\s+/g, " ").trim();
        if (!seen.has(v)){ seen.add(v); out.push(v); }
        if (out.length >= 3) return true;
      }
    }
    return false;
  };
  if (!scan(RE_NUM)) scan(RE_ALT);
  return out;
}

});

  // 按 ISO 周切段（只保留落在本月内的日期）

  for (let day = 1; day <= end.getDate(); day++){

    let w = weeks[weeks.length - 1];

  const wRows = [], eRows = [];
  const total = { files: 0, days: 0, ev: 0, hot: 0 };
  const fill = { 简述: 0, 洞察: 0, 改进: 0, 计划: 0, 推演: 0 };

  for (const w of weeks){
    const covered = w.keys.map(k => `${k.slice(4,6)}/${k.slice(6,8)}`);
    const have = [];
    let evCount = 0, hotCount = 0;

        let nums = pickNums(ev.free["简述"], ev.fields["洞察"], ev.fields["改进"], ev.free["计划"], ev.free["推演"]);
        if (!nums.length) nums = pickNums(ev.free["分析"]);
        const inner = ev.fields["洞察"] || ev.fields["改进"];
        const planText = flat(ev.free["计划"]) || Object.entries(ev.plan).map(([a, b]) => `${a}：${b}`).join("；");
        eRows.push([
          `${k.slice(4,6)}/${k.slice(6,8)}`,
          `**${ev.title}**` + (ev.rating ? ` <small>★${ev.rating}</small>` : "") + (ev.themes.length ? `<br><small>${ev.themes.join(" ")}</small>` : "") + (ev.persons.length ? `<br><small>👤 ${ev.persons.map(p => cut(p, 24)).join(" / ")}</small>` : ""),
          inner ? cut(inner, 64) : MUTE + "未填「洞察 / 改进」</span>",
          planText ? cut(planText, 64) : MUTE + "计划栏为空</span>",

          p.file.link

      }

    total.ev += evCount; total.hot += hotCount;

    const found = wkLink.get(`W${w.iso}`);

## 三、从周报到月报：跨周连接

1. **内在秩序链（event-2026-005）在本月走了两步，但没有走出第三步。** 09/03 在游戏场景里发现「埋怨 = 归因偏差 + 威胁反应」，并给出三问自检；09/14（11 天后）把它上升为「理性是保护内心的屏障」。两处都停在认知层：09/03 的🎯计划与🔮推演为空，09/14 同样为空。**连续两个高分事件都没有执行记录**，这是本月最稳定的模式，而不是偶发。
2. **作息是前置变量，不是并列变量。** 09/15 凌晨 3 点睡 → 12 点起床 → 上午时间全失 → 傍晚再补觉 → 20 点才恢复；09/23 饭后刷短视频 → 昏睡到 8:30 → 少学 1 小时。两次的机制相同：**恢复动作本身吃掉了下一个学习时段**。09/29 没有记录入睡时间，链条在此断档，因此本月无法算出「作息 → 时长」的量化关系，只能说两次样本方向一致。
3. **时间泄漏的清单化带来了当天的回升。** 09/23 把泄漏点写成四类并算出「改良后 +4 小时」，09/24 实际从 5 小时回到 6 小时（底线 3 小时）。这是本月唯一一次「写完改进 → 次日结果变化」的闭环，但缺 4 小时中的 2 小时仍花在杀毒软件上，说明清单解决的是**觉察**，不是**替代方案**。
4. **方法迭代走了两条不同的路：拆任务失败，改流程有效。** 09/23 提「滴答清单四类拆分」，本周没有执行；09/29 换成「中文反译回德语 + 复听」，当天就用掉 4–5 小时并自评效果大幅提升。两者的差别不在努力程度，而在**是否改变了处理材料的动作**。注意：这只是时间相邻，日记没有写出「因为拆分失败所以改流程」的因果。
5. **德语写作在这个月只发生了一次，且没有完成。** 09/15 写《Verbesserung der Lernzeit》（3 条只写 2 条），09/23 明确「没有做到练习作文」。整个 9 月写作输出 = 1 次未完成稿、0 次批改、0 个分数。**任何「9 月德语进步了」的说法，在日记层面都没有数据支撑。**

7. **恋爱 / 社交线从「取舍」变成「并行」，并在 10/01 推进到「改交互结构」。** 04/21 的结论是「先出国」，09/22 记下「被看见 / 被理解 / 被接受」的沟通方法，09/29 直接下载 APP 并设「每天最多半小时」。取舍没变，但多了一个时间成本上限——这是本月唯一一处把「计划」写成可计量约束的地方。10/01 的复盘属于 10 月，不计入本月。
