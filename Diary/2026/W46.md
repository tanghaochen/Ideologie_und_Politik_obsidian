// ══════════════════════════════════════════════════════════════════════
// 周报格子引擎：只定义函数，不生成 Markdown / HTML 表格
// Markdown 表格的每个格子独立调用：`$= WRF(dv).xxx(星期i, 第k条)`
// 异步格子使用 `$= await WRF(dv).xxx(...)`
// ══════════════════════════════════════════════════════════════════════

// 用全局标记避免同一页面重复初始化时覆盖共享缓存。
globalThis.__WRF_CACHE = globalThis.__WRF_CACHE || new Map();

globalThis.WRF = function (dv) {
  const cur = dv.current() || {};
  const active = app.workspace.getActiveFile();
  const clues = [];
  if (cur.file) clues.push(`${cur.file.name || ""} ${cur.file.path || ""}`);
  if (active) clues.push(`${active.basename || ""} ${active.path || ""}`);

  const yearMatch = clues.join(" ").match(/Diary\/(\d{4})\//)
    || clues.join(" ").match(/(20\d{2})/);
  const Y = yearMatch ? Number(yearMatch[1]) : new Date().getFullYear();

  const p2 = n => String(n).padStart(2, "0");
  const isoWeek = date => {
    const d = new Date(Date.UTC(date.getFullYear(), date.getMonth(), date.getDate()));
    d.setUTCDate(d.getUTCDate() + 4 - (d.getUTCDay() || 7));
    const jan1 = new Date(Date.UTC(d.getUTCFullYear(), 0, 1));
    return Math.ceil((((d - jan1) / 86400000) + 1) / 7);
  };

  let WK = 0;
  for (const clue of clues) {
    const byWeek = clue.match(/[Ww](\d{1,2})/);
    if (byWeek) { WK = Number(byWeek[1]); break; }
    const byDate = clue.match(/(20\d{2})\D?(\d{2})\D?(\d{2})(?!\d)/);
    if (byDate) {
      const candidate = new Date(Number(byDate[1]), Number(byDate[2]) - 1, Number(byDate[3]));
      const w = isoWeek(candidate);
      if (w >= 1 && w <= 53) { WK = w; break; }
    }
  }
  if (!(WK >= 1 && WK <= 53)) WK = isoWeek(new Date());

  // ISO 周：以 ISO 年的 1 月 4 日所在周为第 1 周。
  const jan4 = new Date(Y, 0, 4);
  const MON = new Date(jan4);
  MON.setDate(jan4.getDate() - ((jan4.getDay() || 7) - 1) + (WK - 1) * 7);
  MON.setHours(0, 0, 0, 0);

  const dayOf = i => {
    const d = new Date(MON);
    d.setDate(d.getDate() + i);
    return d;
  };
  const ymd = d => `${d.getFullYear()}${p2(d.getMonth() + 1)}${p2(d.getDate())}`;
  const md = d => `${p2(d.getMonth() + 1)}/${p2(d.getDate())}`;
  const fullDate = d => `${d.getFullYear()}-${p2(d.getMonth() + 1)}-${p2(d.getDate())}`;
  const DAY = ["周一", "周二", "周三", "周四", "周五", "周六", "周日"];

  // 跨年周按日期自己的年份去对应 Diary/YYYY/ 目录查找。
  const fileOf = d => {
    const prefix = `Diary/${d.getFullYear()}/`;
    const name = ymd(d);
    return app.vault.getMarkdownFiles().find(f => f.path.startsWith(prefix) && f.basename.startsWith(name));
  };

  const clean = value => String(value ?? "")
    .replace(/%%[\s\S]*?%%/g, "")
    .replace(/<!--[\s\S]*?-->/g, "")
    .replace(/!\[\[[^\]]*\]\]/g, "")
    .replace(/\[\[([^\]|]*)\|([^\]]*)\]\]/g, "$2")
    .replace(/\[\[([^\]]*)\]\]/g, "$1")
    .replace(/^[ \t]*#{1,6}[ \t]*/gm, "")
    .replace(/\*\*/g, "")
    .split("|").join("／")
    .replace(/\s+/g, " ")
    .trim();

  const cut = (value, max = 150) => {
    const text = clean(value);
    return text.length > max ? text.slice(0, max) + "…" : text;
  };

  // 取某个三级标题的正文；只把下一个二级/三级标题视为边界，#### 不截断。
  const sectionOf = (block, name) => {
    const heading = `### ${name}`;
    const at = block.indexOf(heading);
    if (at < 0) return "";
    const rest = block.slice(at + heading.length);
    const next = rest.search(/^#{2,3}\s/m);
    return next < 0 ? rest : rest.slice(0, next);
  };

  const parse = raw => String(raw || "")
    .replace(/%%[\s\S]*?%%/g, "")
    .split(/^##\s*📌\s*/m)
    .slice(1)
    .map(block => {
      const head = (block.split(/\r?\n/)[0] || "").trim();
      const score = (block.match(/^[ \t]*-?[ \t]*重要度::[ \t]*(\d+)/m) || [])[1] || "";

      const analysis = sectionOf(block, "🔍 分析");
      const mind = (analysis.match(/^[ \t]*-?[ \t]*洞察::[ \t]*([^\r\n]*)$/m) || [])[1] || "";

      const plan = sectionOf(block, "🎯 计划");
      const what = (plan.match(/^[ \t]*[-*]?[ \t]*What::[ \t]*([^\r\n]*)$/mi) || [])[1] || "";
      const when = (plan.match(/^[ \t]*[-*]?[ \t]*When::[ \t]*([^\r\n]*)$/mi) || [])[1] || "";
      const act = [clean(what), clean(when)].filter(Boolean).join(" ｜ ");

      let result = sectionOf(block, "📝 简述");
      result = result.replace(/^\s*第[-一二三四五六七八九十\d]+件事[：:]\s*/, "");
      // 若“第 N 件事：”前面有空行，先压平再删前缀。
      result = clean(result).replace(/^第[-一二三四五六七八九十\d]+件事[：:]\s*/, "");

      return { head, score, mind: clean(mind), act, result };
    });

  // 按文件修改时间缓存 Promise：同一天的多个格子共用一次文件读取。
  const load = d => {
    const file = fileOf(d);
    if (!file) return Promise.resolve(null);
    const key = `${file.path}:${file.stat?.mtime || 0}`;
    if (!globalThis.__WRF_CACHE.has(key)) {
      const promise = dv.io.load(file.path).then(raw => parse(raw));
      globalThis.__WRF_CACHE.set(key, promise);
    }
    return globalThis.__WRF_CACHE.get(key);
  };

  return {
    week: () => WK,
    head: () => `W${WK} · ${fullDate(MON)} ~ ${fullDate(dayOf(6))}`,
    span: () => `${fullDate(MON)}（周一）～ ${fullDate(dayOf(6))}（周日）`,
    cover: async () => {
      let days = 0, events = 0;
      for (let i = 0; i < 7; i++) {
        const d = dayOf(i);
        const data = await load(d);
        if (data !== null) days++;
        events += data ? data.length : 0;
      }
      return `${days} / 7 天有日记｜事件 ${events} 条`;
    },
    count: async i => {
      const data = await load(dayOf(i));
      return data ? data.length : 0;
    },
    d: (i, k) => k === 0 ? `${DAY[i]} ${md(dayOf(i))}` : "",
    ev: async (i, k) => {
      const d = dayOf(i);
      const data = await load(d);
      if (data === null) return k === 0 ? "暂无日记记录" : "—";
      if (!data.length) return k === 0 ? "有日记，但没有 📌 事件" : "—";
      const event = data[k];
      if (!event) return "—";
      return dv.sectionLink(`Diary/${d.getFullYear()}/${ymd(d)}`, `📌 ${event.head}`, false, cut(event.head, 150));
    },
    mind: async (i, k) => {
      const data = await load(dayOf(i));
      return data && data[k] ? cut(data[k].mind) || "—" : "—";
    },
    act: async (i, k) => {
      const data = await load(dayOf(i));
      return data && data[k] ? cut(data[k].act) || "—" : "—";
    },
    res: async (i, k) => {
      const data = await load(dayOf(i));
      return data && data[k] ? cut(data[k].result) || "—" : "—";
    },
    notes: async (i, k) => {
      const d = dayOf(i);
      const data = await load(d);
      if (!data || !data[k]) return "—";
      const link = dv.fileLink(`Diary/${d.getFullYear()}/${ymd(d)}`, false, ymd(d));
      return dv.span([link, ` · ★${data[k].score || "?"} · 📌×${data.length}`]);
    }
  };
};

// 隐藏这个“函数库”代码块；它不负责生成任何表格内容。
this.container.style.display = "none";
