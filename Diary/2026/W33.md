```dataviewjs
// ═══════════════ ⚙️ 配置区（每周只改这里） ═══════════════
const startDate = "20260810";   // 本周起始日
const endDate   = "20260822";   // 本周结束日
const folder    = "Diary/2026"; // 日记归档目录
const MAX_CHARS = 150;           // 单条记录最大字数（超出截断）

// ═══════════════ 🎨 类别样式定义 ═══════════════
const ICONS = {
  "🌟": { label: "Keep · 保持", color: "#34a853" },
  "🚩": { label: "Problem · 问题", color: "#ea4335" },
  "💡": { label: "Try · 尝试", color: "#fbbc04" }
};

// ═══════════════ 🔧 注入 CSS：强制换行 + 防撑爆 ═══════════════
const css = document.createElement("style");
css.textContent = `
  .kpt-weekly { width: 100%; border-collapse: collapse; table-layout: fixed; }
  .kpt-weekly th, .kpt-weekly td {
    border: 1px solid var(--background-modifier-border);
    padding: 8px 10px; vertical-align: top;
    white-space: normal !important;        /* 核心：强制换行 */
    word-break: break-word !important;      /* 长词强制断行 */
    overflow-wrap: anywhere !important;
  }
  .kpt-weekly th { background: var(--background-secondary); text-align: center; }
  .kpt-weekly .col-macro { width: 22%; text-align: center; }
  .kpt-weekly .kpt-date  { font-weight: bold; font-size: 1.05em; }
  .kpt-weekly .kpt-link  { font-size: 0.85em; margin-top: 4px; }
  .kpt-weekly .kpt-stats { font-size: 0.8em; opacity: 0.75; margin-top: 4px; }
  .kpt-weekly ul { margin: 0; padding-left: 0; list-style: none; }
  .kpt-weekly li {
    margin: 4px 0; padding: 3px 8px;
    border-left: 4px solid;                 /* 彩色左边框分类 */
    border-radius: 3px;
    background: var(--background-secondary-alt);
    font-size: 0.92em; line-height: 1.5;
    display: -webkit-box; -webkit-box-orient: vertical;
    -webkit-line-clamp: 3; overflow: hidden; /* CSS 双保险：最多 3 行 */
  }
  .kpt-weekly .empty { opacity: 0.4; font-style: italic; }
  .kpt-summary { margin-top: 12px; font-size: 0.95em; }
  .kpt-badge {
    display: inline-block; padding: 2px 12px; margin-right: 8px;
    border-radius: 12px; color: #fff; font-weight: bold; font-size: 0.85em;
  }
`;
this.container.appendChild(css);

// ═══════════════ 📥 抓取并解析日记 ═══════════════
const pages = dv.pages(`"${folder}"`)
  .where(p => /^\d{8}$/.test(p.file.name) && p.file.name >= startDate && p.file.name <= endDate)
  .sort(p => p.file.name);

const WEEK = ["日","一","二","三","四","五","六"];
const esc = s => s.replace(/&/g,"&amp;").replace(/</g,"&lt;").replace(/>/g,"&gt;");
const totals = { "🌟": 0, "🚩": 0, "💡": 0 };
let rows = "";

for (const page of pages) {
  const content = await dv.io.load(page.file.path);
  const items = [];

  for (const line of content.split("\n")) {
    const m = line.match(/^\s*-\s*(🌟|🚩|💡)\s*(.+?)\s*$/);
    if (m) {
      let text = m[2];
      if (text.length > MAX_CHARS) text = text.slice(0, MAX_CHARS) + "…"; // 硬性截断
      items.push({ icon: m[1], text });
      totals[m[1]]++;
    }
  }

  const d = page.file.name;
  const date = new Date(+d.slice(0,4), +d.slice(4,6)-1, +d.slice(6,8));
  const dateStr = `${d.slice(4,6)}/${d.slice(6,8)} 周${WEEK[date.getDay()]}`;

  const listHtml = items.length
    ? "<ul>" + items.map(i =>
        `<li style="border-color:${ICONS[i.icon].color}">${i.icon} ${esc(i.text)}</li>`
      ).join("") + "</ul>"
    : `<span class="empty">— 无复盘记录 —</span>`;

  rows += `
    <tr>
      <td class="col-macro">
        <div class="kpt-date">${dateStr}</div>
        <div class="kpt-link">${page.file.link}</div>
        <div class="kpt-stats">🌟×${items.filter(i=>i.icon==="🌟").length}
          🚩×${items.filter(i=>i.icon==="🚩").length}
          💡×${items.filter(i=>i.icon==="💡").length}</div>
      </td>
      <td>${listHtml}</td>
    </tr>`;
}

// ═══════════════ 📊 渲染表格 + 本周总结 ═══════════════
this.container.innerHTML += `
  <table class="kpt-weekly">
    <thead><tr><th style="width:22%">📅 宏观 · 日期</th><th>🔍 微观 · 每日复盘明细</th></tr></thead>
    <tbody>${rows || `<tr><td colspan="2" class="empty">本周范围内没有找到日记</td></tr>`}</tbody>
  </table>
  <div class="kpt-summary">
    ${Object.keys(ICONS).map(k =>
      `<span class="kpt-badge" style="background:${ICONS[k].color}">${k} ${ICONS[k].label} × ${totals[k]}</span>`
    ).join("")}
  </div>`;
```

```dataviewjs
// ═══════════════ ⚙️ 配置区（每周只改这里） ═══════════════
const startDate = "20260810";
const endDate   = "20260823";
const folder    = "Diary/2026";
const TOP_N     = 3;    // 聚焦最重要 N 件事
const MAX_FREE  = 80;   // 自由文本（简述/推演）截断字数
const MAX_FIELD = 50;   // 字段类内容截断字数

// ═══════════════ 🎨 CSS 注入 ═══════════════
const css = document.createElement("style");
css.textContent = `
  .wk-table{width:100%;border-collapse:collapse;table-layout:fixed;margin-top:10px}
  .wk-table th,.wk-table td{border:1px solid var(--background-modifier-border);
    padding:6px 8px;vertical-align:top;font-size:.88em;line-height:1.5;
    white-space:normal!important;word-break:break-word!important;overflow-wrap:anywhere!important}
  .wk-table th{background:var(--background-secondary)}
  .wk-grid{display:flex;gap:10px;flex-wrap:wrap;margin:8px 0}
  .wk-card{flex:1;min-width:230px;border:1px solid var(--background-modifier-border);
    border-top:4px solid var(--interactive-accent);border-radius:8px;
    padding:10px;background:var(--background-secondary-alt)}
  .wk-card .t{font-weight:bold;font-size:1.02em;margin:3px 0}
  .wk-card .s{color:var(--text-accent);letter-spacing:2px}
  .wk-card .d{font-size:.78em;opacity:.65}
  .wk-card .sec{font-size:.85em;margin-top:6px}
  .wk-lbl{font-weight:bold;font-size:.78em;opacity:.8}
  .wk-tag{display:inline-block;background:var(--interactive-accent);
    color:var(--text-on-accent);border-radius:10px;padding:1px 8px;
    font-size:.72em;margin:2px 3px 2px 0}
  .wk-chip{display:inline-block;border:1px dashed var(--text-muted);
    border-radius:6px;padding:1px 6px;font-size:.75em;margin:2px 3px 2px 0}
  .wk-h{font-size:1.1em;font-weight:bold;margin:18px 0 6px;
    padding-left:8px;border-left:4px solid var(--interactive-accent)}
  .wk-empty{opacity:.4;font-style:italic}
  .wk-badge{display:inline-block;padding:2px 12px;margin-right:8px;border-radius:12px;
    background:var(--background-secondary);font-size:.85em;font-weight:bold}
`;
this.container.appendChild(css);

// ═══════════════ 🔧 工具函数 ═══════════════
const esc = s => String(s).replace(/&/g,"&amp;").replace(/</g,"&lt;").replace(/>/g,"&gt;");
const cut = (s,n) => !s ? "" : (s.length > n ? s.slice(0,n)+"…" : s);
const WEEK = ["日","一","二","三","四","五","六"];
const dateLabel = d => `${d.slice(4,6)}/${d.slice(6,8)} 周${WEEK[new Date(+d.slice(0,4),+d.slice(4,6)-1,+d.slice(6,8)).getDay()]}`;
const stars = r => "★".repeat(r) + "☆".repeat(5-r);
const alink = (l,text) => `<a href="${l.path}" data-href="${l.path}" class="internal-link">${esc(text)}</a>`;

// ═══════════════ 📥 块级解析器：以“事件”为单位 ═══════════════
function parseDiary(content, page){
  const events = [];
  const blocks = content.split(/^## 📌\s*/m).slice(1); // 按事件标题切块
  for (const b of blocks){
    const lines = b.split("\n");
    const ev = { title:lines[0].trim(), date:page.file.name, link:page.file.link,
                 rating:0, themes:[], persons:[], plan:{}, fields:{}, free:{} };
    let section = null;
    for (const line of lines.slice(1)){
      const h = line.match(/^###\s*(.+)/);
      if (h){ section = h[1].replace(/[^一-龥A-Za-z]/g,""); continue; }
      const f = line.match(/^\s*-?\s*([\w一-龥]+)::\s*(.+)/);
      if (f){
        const k=f[1], v=f[2].trim();
        if (k==="重要度") ev.rating = Math.min(5, parseInt(v)||0);
        else if (k==="主题") ev.themes = v.split(/[\s,，]+/).filter(t=>t.startsWith("#"));
        else if (k==="人物") ev.persons.push(v);
        else if (["What","Who","When","Where","How","预算"].includes(k)) ev.plan[k]=v;
        else ev.fields[k] = ev.fields[k] ? ev.fields[k]+"；"+v : v; // 洞察/改进等，多条合并
      } else if (section && line.trim()){
        ev.free[section] = (ev.free[section]||"") + line.trim() + " "; // 自由长文拼接
      }
    }
    events.push(ev);
  }
  return events;
}

let all = [];
for (const p of dv.pages(`"${folder}"`)
    .where(p=>/^\d{8}$/.test(p.file.name) && p.file.name>=startDate && p.file.name<=endDate)
    .sort(p=>p.file.name)){
  all.push(...parseDiary(await dv.io.load(p.file.path), p));
}

// ═══════════════ 🧠 周级智能聚合 ═══════════════
// 1. 聚焦三件事：按重要度排序
const top = [...all].sort((a,b)=>b.rating-a.rating).slice(0,TOP_N);

// 2. 关联发现：标签 → 事件 反查表
const tagMap = {};
all.forEach(e=>e.themes.forEach(t=>(tagMap[t]=tagMap[t]||[]).push(e)));
const related = Object.entries(tagMap).filter(([t,es])=>es.length>=2)
                .sort((a,b)=>b[1].length-a[1].length);
const sharedInTop = top.length>1
  ? top[0].themes.filter(t=>top.every(e=>e.themes.includes(t))) : [];
const mainLine = Object.entries(tagMap).sort((a,b)=>b[1].length-a[1].length)[0];

// 3. 人物档案：跨日记聚合同一人物的所有观察
const personMap = {};
all.forEach(e=>e.persons.forEach(p=>{
  const i = p.search(/[—–\-:：]/);
  const name = i>0 ? p.slice(0,i).trim() : p.trim();
  const note = i>0 ? p.slice(i+1).trim() : "";
  (personMap[name]=personMap[name]||[]).push({note, ev:e});
}));

// 4. 预算汇总
let money=0, hours=0;
all.forEach(e=>{ const b=e.plan["预算"]||"";
  const m=b.match(/([\d.]+)\s*元/), h=b.match(/([\d.]+)\s*小时/);
  if(m) money+=+m[1]; if(h) hours+=+h[1];
});

// ═══════════════ 🖼️ 渲染 ═══════════════
let html = "";

// —— 面板①：本周聚焦 TOP N ——
html += `<div class="wk-h">🎯 本周聚焦：最重要的 ${TOP_N} 件事</div><div class="wk-grid">`;
html += top.map(e=>`
  <div class="wk-card">
    <div class="d">${dateLabel(e.date)}</div>
    <div class="t">${alink(e.link, e.title)}</div>
    <div class="s">${stars(e.rating)}</div>
    <div>${e.themes.map(t=>`<span class="wk-tag">${t}</span>`).join("")}</div>
    <div class="sec"><span class="wk-lbl">简述｜</span>${esc(cut((e.free["简述"]||"").trim(),MAX_FREE))||"—"}</div>
    ${e.fields["洞察"]?`<div class="sec"><span class="wk-lbl">💡洞察｜</span>${esc(cut(e.fields["洞察"],MAX_FIELD))}</div>`:""}
    ${e.fields["改进"]?`<div class="sec"><span class="wk-lbl">🔧改进｜</span>${esc(cut(e.fields["改进"],MAX_FIELD))}</div>`:""}
    ${e.plan["When"]?`<div class="sec"><span class="wk-lbl">⏰</span> ${esc(e.plan["When"])}${e.plan["预算"]?` ｜💰${esc(e.plan["预算"])}`:""}</div>`:""}
  </div>`).join("") || `<span class="wk-empty">本周无事件</span>`;
html += `</div>`;

// 三件事的关联提示（导航）
html += `<div>`;
if (sharedInTop.length)
  html += `<span class="wk-badge">🔗 三件事的共同主线：${sharedInTop.join("、")}</span>`;
if (mainLine)
  html += `<span class="wk-badge">🧭 本周最高频主题：${mainLine[0]}（${mainLine[1].length} 件事）</span>`;
html += `<span class="wk-badge">📌 事件 ${all.length} 件</span>
         <span class="wk-badge">💰 计划预算 ${money}元 / ${hours}小时</span></div>`;

// —— 面板②：关联导航 ——
html += `<div class="wk-h">🔗 关联导航（同一主题下的事件簇）</div>`;
html += related.length
  ? related.map(([t,es])=>`<div style="margin:4px 0"><span class="wk-tag">${t}</span> → ${
      es.map(e=>`${dateLabel(e.date)} ${alink(e.link,e.title)}`).join(" ｜ ")}</div>`).join("")
  : `<span class="wk-empty">本周事件之间暂无共同主题，试试统一打标签</span>`;

// —— 面板③：人物档案 ——
const pEntries = Object.entries(personMap);
if (pEntries.length){
  html += `<div class="wk-h">👥 本周人物观察档案</div>
  <table class="wk-table"><thead><tr>
    <th style="width:14%">人物</th><th style="width:8%">提及</th><th>观察（在意/优势/思维习惯）</th><th style="width:22%">相关事件</th>
  </tr></thead><tbody>`;
  html += pEntries.map(([name,arr])=>`<tr>
    <td><b>${esc(name)}</b></td><td style="text-align:center">×${arr.length}</td>
    <td>${arr.map(a=>esc(cut(a.note,MAX_FIELD))).filter(Boolean).join("；")}</td>
    <td>${arr.map(a=>alink(a.ev.link,a.ev.title)).join("<br>")}</td>
  </tr>`).join("");
  html += `</tbody></table>`;
}

// —— 面板④：全量事件明细 ——
html += `<div class="wk-h">📋 全部事件明细</div>
<table class="wk-table"><thead><tr>
  <th style="width:10%">日期</th><th style="width:18%">事件 / 重要度</th>
  <th style="width:24%">📝 简述</th><th style="width:24%">🔍 分析</th><th>🎯 计划 & 推演</th>
</tr></thead><tbody>`;
html += all.map(e=>{
  const planHtml = Object.entries(e.plan)
    .map(([k,v])=>`<span class="wk-chip"><b>${k}</b> ${esc(cut(v,28))}</span>`).join("");
  const anaHtml = Object.entries(e.fields)
    .map(([k,v])=>`<div><span class="wk-lbl">${k}｜</span>${esc(cut(v,MAX_FIELD))}</div>`).join("");
  return `<tr>
    <td>${dateLabel(e.date)}</td>
    <td><b>${alink(e.link,e.title)}</b><br><span class="s">${stars(e.rating)}</span><br>
        ${e.themes.map(t=>`<span class="wk-tag">${t}</span>`).join("")}</td>
    <td>${esc(cut((e.free["简述"]||"").trim(),MAX_FREE))||'<span class="wk-empty">—</span>'}</td>
    <td>${e.persons.map(p=>`<div>👤 ${esc(cut(p,MAX_FIELD))}</div>`).join("")}${anaHtml}</td>
    <td>${planHtml}<div style="margin-top:4px;opacity:.85">${esc(cut((e.free["推演"]||"").trim(),MAX_FREE))}</div></td>
  </tr>`;
}).join("");
html += `</tbody></table>`;
this.container.innerHTML += html;
```
